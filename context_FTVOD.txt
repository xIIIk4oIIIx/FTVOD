object NetworkModule {
    private const val BASE_URL = "https://filman.cc"
    // ZMIANA: User-Agent identyczny z HeadlessLoginProvider (Chrome 124)
    private const val TV_USER_AGENT = "Mozilla/5.0 (Linux; Android 10; K) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Mobile Safari/537.36"

    private var clientInstance: OkHttpClient? = null

    fun getHttpClient(context: Context): OkHttpClient {
        if (clientInstance == null) {
            val trustAllCerts = arrayOf<javax.net.ssl.TrustManager>(object : javax.net.ssl.X509TrustManager {
                override fun checkClientTrusted(chain: Array<out java.security.cert.X509Certificate>?, authType: String?) {}
                override fun checkServerTrusted(chain: Array<out java.security.cert.X509Certificate>?, authType: String?) {}
                override fun getAcceptedIssuers(): Array<java.security.cert.X509Certificate> = arrayOf()
            })

            val sslContext = javax.net.ssl.SSLContext.getInstance("SSL")
            sslContext.init(null, trustAllCerts, java.security.SecureRandom())

            clientInstance = OkHttpClient.Builder()
                .sslSocketFactory(sslContext.socketFactory, trustAllCerts[0] as javax.net.ssl.X509TrustManager)
                .hostnameVerifier { _, _ -> true }
                .connectTimeout(20, TimeUnit.SECONDS)
                .readTimeout(60, TimeUnit.SECONDS)
                .addInterceptor { chain ->
                    val prefs = context.getSharedPreferences("filman_data", Context.MODE_PRIVATE)
                    val cookies = prefs.getString("cookies", null)

                    val requestBuilder = chain.request().newBuilder()
                        .header("User-Agent", TV_USER_AGENT)
                        .header("Referer", BASE_URL)

                    if (!cookies.isNullOrEmpty()) {
                        requestBuilder.header("Cookie", cookies)
                    }

                    val response = chain.proceed(requestBuilder.build())

                    // Sprawdzenie czy ostateczny URL przekierował do logowania
                    if (response.request.url.toString().endsWith("/logowanie")) {
                        return@addInterceptor response.newBuilder()
                            .code(401)
                            .message("Unauthorized")
                            .build()
                    }

                    response
                }
                .build()
        }
        return clientInstance!!
    }
}

package com.k4o.filmantv.data

import android.content.Context
import androidx.core.content.edit

/**
 * FavoritesManager - Klasa odpowiedzialna za trwałe przechowywanie listy ulubionych
 * filmów i seriali użytkownika w pamięci lokalnej urządzenia.
 */
class FavoritesManager(context: Context) {
    private val prefs = context.getSharedPreferences("filman_favorites", Context.MODE_PRIVATE)

    /**
     * Dodaje film lub serial do ulubionych.
     * Zapisuje metadane pozwalające na odtworzenie obiektu MovieItem bez dostępu do sieci.
     */
    fun addFavorite(movie: MovieItem) {
        val url = movie.originalUrl
        prefs.edit {
            // Główny klucz istnienia
            putBoolean(url, true)
            // Metadane
            putString("${url}_title", movie.title)
            putString("${url}_thumb", movie.thumbnail)
            putString("${url}_type", movie.mediaType)
            // Timestamp do sortowania (najnowsze na górze)
            putLong("${url}_timestamp", System.currentTimeMillis())
        }
    }

    /**
     * Usuwa element z listy ulubionych oraz czyści powiązane metadane.
     */
    fun removeFavorite(url: String) {
        prefs.edit {
            remove(url)
            remove("${url}_title")
            remove("${url}_thumb")
            remove("${url}_type")
            remove("${url}_timestamp")
        }
    }

    /**
     * Sprawdza, czy dany materiał (identyfikowany przez URL) jest w ulubionych.
     */
    fun isFavorite(url: String): Boolean {
        return prefs.getBoolean(url, false)
    }

    /**
     * Pobiera pełną listę ulubionych produkcji.
     * Wynik jest sortowany malejąco według czasu dodania.
     */
    fun getFavorites(): List<MovieItem> {
        val allKeys = prefs.all.keys

        // Filtrowanie kluczy bazowych (ignorujemy klucze metadanych z sufiksami)
        val movieUrls = allKeys.filter {
            !it.endsWith("_title") &&
                    !it.endsWith("_thumb") &&
                    !it.endsWith("_type") &&
                    !it.endsWith("_timestamp") &&
                    prefs.all[it] is Boolean // Dodatkowe sprawdzenie typu
        }

        return movieUrls.mapNotNull { url ->
            val title = prefs.getString("${url}_title", null) ?: return@mapNotNull null
            val thumb = prefs.getString("${url}_thumb", null)
            val type = prefs.getString("${url}_type", "movie")
            val timestamp = prefs.getLong("${url}_timestamp", 0L)

            MovieItem(
                originalUrl = url,
                title = title,
                year = null,
                thumbnail = thumb,
                overview = null,
                mediaType = type,
                backdropUrl = null,
                genres = emptyList(),
                progress = 0f
            ) to timestamp
        }
            .sortedByDescending { it.second } // Sortowanie: najnowsze na początku
            .map { it.first }
    }
}

package com.k4o.filmantv.data

import android.util.Log
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.OkHttpClient
import okhttp3.Request
import org.jsoup.Jsoup
import org.jsoup.nodes.Document
import org.jsoup.nodes.Element

/**
 * Model szczegółów materiału wideo wraz z linkami.
 */
data class MovieDetails(
    val description: String?,
    val backdropUrl: String?,
    val links: List<StreamItem>
)

/**
 * Wyjątki rzucane przez scraper podczas interakcji z serwisem.
 */
sealed class ScraperException(val msg: String) : Exception(msg) {
    class RateLimitException(message: String = "Wykryto ochronę anty-botową. Spróbuj później") : ScraperException(message)
    class AuthException(message: String) : ScraperException(message)
    class NetworkException(message: String) : ScraperException(message)
}

/**
 * FilmanScraper - Silnik parsujący dane bezpośrednio z serwisu Filman.cc.
 */
class FilmanScraper(
    private val client: OkHttpClient,
    private val headlessSearch: HeadlessSearchProvider
) {

    private val baseUrl = "https://filman.cc"
    private val blockedHosts = listOf("veev", "savefiles", "ups2up", "doodstream", "luluv")

    /**
     * Pobiera listę filmów i seriali ze strony głównej.
     */
    suspend fun getHomePage(): List<MovieItem> = fetchAndParse(baseUrl)

    /**
     * Pobiera listę filmów z podstrony /filmy z obsługą paginacji, gatunku, sortowania i rocznika.
     */
    suspend fun getMovies(
        page: Int,
        genreId: Int?,
        sort: String? = null,
        year: String? = null
    ): List<MovieItem> {
        val urlBuilder = StringBuilder("$baseUrl/filmy/")
        if (genreId != null) {
            urlBuilder.append("category:$genreId/")
        }
        if (sort != null) {
            urlBuilder.append("$sort/")
        }
        if (year != null) {
            urlBuilder.append("year:$year/")
        }
        urlBuilder.append("?page=$page")
        return fetchAndParse(urlBuilder.toString())
    }

    /**
     * Pobiera listę seriali z podstrony /seriale z obsługą paginacji, gatunku, sortowania i rocznika.
     */
    suspend fun getSeries(
        page: Int,
        genreId: Int?,
        sort: String? = null,
        year: String? = null
    ): List<MovieItem> {
        val urlBuilder = StringBuilder("$baseUrl/seriale/")
        if (genreId != null) {
            urlBuilder.append("category:$genreId/")
        }
        if (sort != null) {
            urlBuilder.append("$sort/")
        }
        if (year != null) {
            urlBuilder.append("year:$year/")
        }
        urlBuilder.append("?page=$page")
        return fetchAndParse(urlBuilder.toString())
    }

    /**
     * Wykonuje globalne wyszukiwanie w serwisie z użyciem HeadlessSearchProvider.
     */
    suspend fun search(query: String): List<MovieItem> {
        val encodedQuery = withContext(Dispatchers.IO) {
            java.net.URLEncoder.encode(query, "UTF-8")
        }
        val url = "$baseUrl/search?phrase=$encodedQuery"
        val html = headlessSearch.fetchSearchHtml(url)
            ?: throw ScraperException.NetworkException("Brak danych HTML z HeadlessSearch.")
        
        return parseSearchHtml(html)
    }

    /**
     * Parsuje HTML wyników wyszukiwania.
     */
    fun parseSearchHtml(html: String): List<MovieItem> {
        val doc = Jsoup.parse(html, baseUrl)
        // ZMIANA: Usunięto .item z selektora, aby uniknąć parsowania elementów ze strony głównej przy błędzie.
        val elements = doc.select("#item-list > div, #owl-slider .slide, .film-item")
        return elements.mapNotNull { parseMovieItem(it) }
    }

    /**
     * Pobiera listę odcinków dla danego serialu.
     * Miniaturki są ustawiane na null, aby wymusić fallback do plakatu serialu w UI.
     */
    suspend fun getSeriesEpisodes(url: String): List<MovieItem> = withContext(Dispatchers.IO) {
        val request = Request.Builder().url(url).build()

        try {
            client.newCall(request).execute().use { response ->
                if (response.code == 401) throw ScraperException.AuthException("Wymagane zalogowanie.")

                val html = response.body?.string() ?: return@withContext emptyList()
                val doc = Jsoup.parse(html, url)

                // Wybieramy element nadrzędny li, aby pobrać dane odcinka
                doc.select("#episode-list > li > ul > li").mapNotNull { li ->
                    val a = li.select("a").first() ?: return@mapNotNull null
                    val originalUrl = a.attr("abs:href")

                    if (originalUrl.isEmpty() || originalUrl.endsWith("/#")) return@mapNotNull null

                    val title = a.text()

                    MovieItem(
                        originalUrl = originalUrl,
                        title = title,
                        year = null,
                        thumbnail = null, // Uproszczenie: UI użyje fallbackThumbnail
                        overview = null,
                        mediaType = "episode",
                        backdropUrl = null,
                        genres = emptyList(),
                        progress = 0f
                    )
                }
            }
        } catch (e: Exception) {
            if (e is ScraperException) throw e
            Log.e("FilmanScraper", "Błąd pobierania odcinków: $url", e)
            emptyList()
        }
    }

    /**
     * Pobiera szczegóły filmu/odcinka (opis, tło) oraz listę dostępnych serwerów.
     */
    suspend fun getMovieDetails(url: String): MovieDetails = withContext(Dispatchers.IO) {
        val request = Request.Builder().url(url).build()

        try {
            client.newCall(request).execute().use { response ->
                if (response.code == 401) throw ScraperException.AuthException("Wymagane zalogowanie.")

                val html = response.body?.string() ?: return@withContext MovieDetails(null, null, emptyList())
                val doc = Jsoup.parse(html, url)

                // 1. Parsowanie opisu
                val description = doc.select(".film-plot, .description, #opis, [itemprop=description]").first()?.text()
                    ?: doc.select("meta[property=og:description]").attr("content").ifEmpty { "Brak opisu" }

                // 2. Parsowanie tła (Backdrop)
                var backdrop = doc.select("meta[property=og:image]").attr("content").ifEmpty { null }

                // Próba znalezienia tła w stylach inline jeśli og:image to plakat
                val styleAttr = doc.select(".item-header, #item-details").attr("style")
                if (styleAttr.contains("background-image")) {
                    val regex = Regex("""url\(['"]?([^'"]+)['"]?\)""")
                    val match = regex.find(styleAttr)
                    if (match != null) {
                        backdrop = match.groupValues[1]
                    }
                }

                // 3. Parsowanie linków
                val links = doc.select("#links tbody tr").mapNotNull { tr ->
                    val hostImg = tr.select("td:nth-child(1) img").first()
                    val hostName = hostImg?.attr("alt") ?: tr.select("td:nth-child(1)").text()

                    // Filtracja zablokowanych hostów
                    if (blockedHosts.any { hostName.contains(it, ignoreCase = true) }) {
                        return@mapNotNull null
                    }

                    val language = tr.select("td:nth-child(2)").text().trim()
                    val quality = tr.select("td:nth-child(3)").text().trim()

                    val linkElement = tr.select("a[data-iframe]").first() ?: return@mapNotNull null

                    try {
                        val encoded = linkElement.attr("data-iframe")
                        val jsonStr = String(android.util.Base64.decode(encoded, android.util.Base64.DEFAULT))
                        val src = org.json.JSONObject(jsonStr).getString("src")
                        StreamItem(
                            name = hostName,
                            quality = quality,
                            url = src,
                            language = language.ifEmpty { "Oryginał" }
                        )
                    } catch (e: Exception) {
                        null
                    }
                }

                MovieDetails(
                    description = description,
                    backdropUrl = backdrop,
                    links = links
                )
            }
        } catch (e: Exception) {
            if (e is ScraperException) throw e
            Log.e("FilmanScraper", "Błąd pobierania szczegółów: $url", e)
            MovieDetails(null, null, emptyList())
        }
    }

    private suspend fun fetchAndParse(url: String): List<MovieItem> = withContext(Dispatchers.IO) {
        val request = Request.Builder().url(url).build()

        try {
            client.newCall(request).execute().use { response ->
                if (response.code == 429 || response.code == 403) {
                    throw ScraperException.RateLimitException()
                }

                if (response.code == 401) {
                    throw ScraperException.AuthException("Wymagane zalogowanie do serwisu.")
                }

                if (!response.isSuccessful) {
                    throw ScraperException.NetworkException("Błąd serwera: ${response.code}")
                }

                val html = response.body?.string() ?: throw ScraperException.NetworkException("Brak danych w odpowiedzi")
                val doc: Document = Jsoup.parse(html, baseUrl)
                val pageTitle = doc.title()

                if (pageTitle.contains("Just a moment", ignoreCase = true) ||
                    pageTitle.contains("Cloudflare", ignoreCase = true)) {
                    throw ScraperException.RateLimitException()
                }

                if (pageTitle.contains("Logowanie", ignoreCase = true) ||
                    doc.select("form[action*='logowanie']").isNotEmpty()) {
                    throw ScraperException.AuthException("Wymagane zalogowanie do serwisu.")
                }

                val elements = doc.select("#item-list > div, #owl-slider .slide, .film-item")
                if (elements.isEmpty()) return@withContext emptyList<MovieItem>()

                elements.mapNotNull { element ->
                    parseMovieItem(element)
                }
            }
        } catch (e: Exception) {
            if (e is ScraperException) throw e
            Log.e("FilmanScraper", "Błąd podczas scrapowania: $url", e)
            throw ScraperException.NetworkException("Wystąpił błąd sieciowy.")
        }
    }

    private fun parseMovieItem(element: Element): MovieItem? {
        val a = element.select(".poster a").first() ?: element.select("a").first()
        val img = element.select(".poster img").first() ?: element.select("img").first()

        val originalUrl = a?.attr("abs:href") ?: ""
        if (originalUrl.isEmpty() || originalUrl.endsWith("/#")) return null

        val title = element.select(".film_title").text().ifEmpty { null }
            ?: img?.attr("alt")?.ifEmpty { null }
            ?: img?.attr("title")?.ifEmpty { null }
            ?: a?.attr("title")
            ?: ""

        if (title.isEmpty()) return null

        val year = element.select(".film_year").text().ifEmpty { null }

        var thumbUrl = img?.attr("data-src")?.ifEmpty { null }
            ?: img?.attr("data-original")?.ifEmpty { null }
            ?: img?.attr("src")
            ?: ""

        if (thumbUrl.isNotEmpty()) {
            thumbUrl = thumbUrl.trim()
            if (thumbUrl.startsWith("/")) {
                thumbUrl = "$baseUrl$thumbUrl"
            } else if (!thumbUrl.startsWith("http")) {
                thumbUrl = "$baseUrl/$thumbUrl"
            }
        }

        val mediaType = if (originalUrl.contains("/serial/") || originalUrl.contains("/s/")) "series" else "movie"

        val genreText = element.select(".film_genre, .genre, .film-genre").text()
        val genres = if (genreText.isNotBlank()) {
            genreText.split(",").map { it.trim() }.filter { it.isNotEmpty() }
        } else {
            listOf("Na czasie")
        }

        return MovieItem(

/**
 * Rozszerzenie z mechanizmem ponawiania prób, przydatne gdy komponenty nie są jeszcze w pełni gotowe w drzewie Compose.
 */
fun FocusRequester.smartRequestFocus(maxRetries: Int = 10) {
    var retries = 0
    fun tryFocus() {
        runCatching {
            this.requestFocus()
        }.onFailure { e ->
            if (retries < maxRetries) {
                retries++
                // Używamy post, aby poczekać na następną klatkę renderowania
                android.os.Handler(android.os.Looper.getMainLooper()).postDelayed({
                    tryFocus()
                }, 100) // Zwiększone do 100ms dla lepszej stabilności
            } else {
                Log.w("FocusSystem", "FocusRequester nie zainicjalizowany po $maxRetries próbach: ${e.message}")
            }
        }
    }
    tryFocus()
}

/**
 * Wrapper zachowany dla kompatybilności, teraz korzystający z mechanizmu smartRequestFocus.
 */
fun FocusRequester.safeRequestFocus() {
    this.smartRequestFocus()
}

import android.content.Context

class PlaybackManager(context: Context) {
    private val prefs = context.getSharedPreferences("strumyczek_playback", Context.MODE_PRIVATE)

    /**
     * Zapisuje postęp dla danego URL wraz z metadanymi filmu dla historii.
     * @param url Unikalny link do filmu (używamy go jako ID)
     * @param title Tytuł filmu/odcinka
     * @param thumbnail Link do miniaturki
     * @param mediaType Typ mediów (movie/series/episode)
     * @param position Czas w milisekundach
     * @param duration Całkowity czas trwania filmu
     */
    fun saveProgress(url: String, title: String?, thumbnail: String?, mediaType: String?, position: Long, duration: Long) {
        // Nie zapisuj, jeśli to sam początek (mniej niż 10 sekund) lub brak duration
        if (position < 10_000 || duration <= 0) return

        val percentage = position.toFloat() / duration.toFloat()

        prefs.edit {
            putLong(url, position)
            putFloat("${url}_progress", percentage)
            if (title != null) putString("${url}_title", title)
            if (thumbnail != null) putString("${url}_thumb", thumbnail)
            if (mediaType != null) putString("${url}_type", mediaType)
            putLong("${url}_timestamp", System.currentTimeMillis())
        }
    }

    /**
     * Pobiera zapisany postęp czasowy. Zwraca 0, jeśli brak zapisu.
     */
    fun getProgress(url: String): Long {
        return prefs.getLong(url, 0L)
    }

    /**
     * Pobiera procentowy postęp odtwarzania (0.0 - 1.0).
     */
    fun getProgressPercentage(url: String): Float {
        return prefs.getFloat("${url}_progress", 0f)
    }

    /**
     * Czyści postęp (np. po obejrzeniu do końca)
     */
    fun clearProgress(url: String) {
        prefs.edit {
            remove(url)
            remove("${url}_progress")
            remove("${url}_title")
            remove("${url}_thumb")
            remove("${url}_type")
            remove("${url}_timestamp")
        }
    }

    /**
     * Zwraca zestaw URLi (kluczy), które mają zapisany postęp.
     */
    fun getActiveMovieUrls(): Set<String> {
        return prefs.all.keys.filter {
            !it.endsWith("_progress") &&
                    !it.endsWith("_title") &&
                    !it.endsWith("_thumb") &&
                    !it.endsWith("_type") &&
                    !it.endsWith("_timestamp")
        }.toSet()
    }

    /**
     * Pobiera listę filmów z historii na podstawie zapisanych metadanych,
     * uniezależniając ją od aktualnie załadowanego katalogu.
     */
    fun getHistoryItems(): List<MovieItem> {
        val urls = getActiveMovieUrls()
        return urls.mapNotNull { url ->
            val title = prefs.getString("${url}_title", null) ?: return@mapNotNull null
            val thumb = prefs.getString("${url}_thumb", null)
            val type = prefs.getString("${url}_type", "movie")

            // Ukrywamy surowe odcinki w historii, by uniknąć duplikatów i braku okładek
            if (type == "episode") return@mapNotNull null

            val progress = prefs.getFloat("${url}_progress", 0f)
            val timestamp = prefs.getLong("${url}_timestamp", 0L)

            MovieItem(
                originalUrl = url,
                title = title,
                year = null,
                thumbnail = thumb,
                overview = null,
                mediaType = type,
                backdropUrl = null,
                genres = emptyList(),
                progress = progress
            ) to timestamp
        }
            .sortedByDescending { it.second }
            .map { it.first }
    }
}

                            el.remove();
                        } else if (!video) {
                            el.remove();
                        }
                    });
                });
                // C. Klikanie przycisków "Close", "Continue", "Zamknij", "Kontynuuj", "X"
                document.querySelectorAll('button, a, div, span').forEach(el => {
                    var text = (el.innerText || el.textContent || "").trim();
                    if (['Close', 'Continue', 'Zamknij', 'Kontynuuj', 'X'].includes(text)) {
                        el.click();
                    }
                });
                // D. Wymuszanie layoutu środkowego
                if (!video) {
                    var x = window.innerWidth / 2;
                    var y = window.innerHeight / 2;
                    document.elementFromPoint(x, y)?.click();
                }
            }, 1000);
        })()
    """.trimIndent()

    /**
     * Wyodrębnia bezpośredni URL (.mp4 / .m3u8 / master.json) z podanego linku embed.
     */
    @SuppressLint("SetJavaScriptEnabled")
    suspend fun extractVideoUrl(embedUrl: String): String? = withTimeoutOrNull(25000L) {
        val prefs = context.getSharedPreferences("filman_data", Context.MODE_PRIVATE)
        val cookies = prefs.getString("cookies", null)

        withContext(Dispatchers.Main) {
            suspendCancellableCoroutine { continuation ->
                val webView = WebView(context)
                var isResumed = false
                var isExtractionFinished = false

                fun resumeOnce(url: String) {
                    if (!isResumed && continuation.isActive) {
                        isResumed = true
                        isExtractionFinished = true
                        Log.d("StreamExtractor", "SUCCESS: Found stream URL: $url")
                        continuation.resume(url)

                        CoroutineScope(Dispatchers.Main).launch {
                            try {
                                webView.evaluateJavascript("window.stopSniffing = true; window.stop();", null)
                                webView.onPause()
                                webView.stopLoading()
                                webView.loadUrl("about:blank")
                                webView.clearHistory()
                                webView.removeAllViews()
                                webView.destroy()
                            } catch (e: Exception) {
                                Log.w("StreamExtractor", "WebView cleanup failed: ${e.message}")
                            }
                        }
                    }
                }

                webView.settings.apply {
                    javaScriptEnabled = true
                    userAgentString = userAgent
                    domStorageEnabled = true
                    @Suppress("DEPRECATION")
                    databaseEnabled = true
                    mediaPlaybackRequiresUserGesture = false
                    safeBrowsingEnabled = false
                    javaScriptCanOpenWindowsAutomatically = false
                    mixedContentMode = WebSettings.MIXED_CONTENT_ALWAYS_ALLOW
                }

                webView.webViewClient = object : WebViewClient() {
                    override fun shouldInterceptRequest(
                        view: WebView?,
                        request: WebResourceRequest?
                    ): WebResourceResponse? {
                        if (isExtractionFinished) return WebResourceResponse("text/plain", "utf-8", null)

                        val url = request?.url.toString().lowercase()

                        // 1. BLOKOWANIE SEGMENTÓW I ŚMIECI (Performance Fix)
                        if (url.contains(".ts") || url.contains("/seg-") || url.contains("segment")) {
                            return WebResourceResponse("text/plain", "utf-8", null)
                        }

                        // 2. DETEKCJA STRUMIENIA (Sukces)
                        if (url.contains(".m3u8") || url.contains(".mp4") || url.contains("master.json")) {
                            if (!url.contains("favicon") && !url.contains(".js") && !url.contains(".ts")) {
                                resumeOnce(request?.url.toString())
                            }
                        }

                        // 3. FILTROWANIE REKLAM I POPUNDERÓW (Blacklist)
                        val adKeywords = listOf("ads", "popunder", "banner", "doubleclick", "onclick", "prebid")

                        if (url.contains(".js") && (url.contains("google") || url.contains("metrika"))) {
                            return super.shouldInterceptRequest(view, request)
                        }

                        if (adKeywords.any { url.contains(it) }) {
                            return WebResourceResponse("text/plain", "utf-8", null)
                        }

                        // 4. OPTYMALIZACJA (Blokowanie ciężkich zasobów graficznych i czcionek)
                        if (url.endsWith(".jpg") || url.endsWith(".png") || url.endsWith(".woff") || url.endsWith(".woff2") || url.endsWith(".ttf")) {
                            return WebResourceResponse("text/plain", "utf-8", null)
                        }

                        return super.shouldInterceptRequest(view, request)
                    }

                    override fun onPageFinished(view: WebView?, url: String?) {
                        super.onPageFinished(view, url)
                        if (!isExtractionFinished) {
                            Log.d("StreamExtractor", "Page finished loading: $url. Injecting Stealth Clicker...")
                            view?.evaluateJavascript(snifferScript, null)
                        }
                    }

                    override fun onReceivedSslError(view: WebView?, handler: SslErrorHandler?, error: SslError?) {
                        Log.w("StreamExtractor", "Bypass SSL for: ${error?.url}")
                        handler?.proceed()
                    }
                }

                // Konfiguracja menedżera ciasteczek
                val cookieManager = CookieManager.getInstance()
                cookieManager.setAcceptCookie(true)
                cookieManager.setAcceptThirdPartyCookies(webView, true)

                if (cookies != null) {
                    cookies.split(";").forEach { cookie ->
                        cookieManager.setCookie("https://filman.cc", cookie.trim())
                    }
                    cookieManager.flush()
                }

                Log.d("StreamExtractor", "Starting extraction process for: $embedUrl")
                webView.loadUrl(embedUrl)

                continuation.invokeOnCancellation {
                    CoroutineScope(Dispatchers.Main).launch {
                        try {
                            webView.evaluateJavascript("window.stopSniffing = true; window.stop();", null)
                            webView.onPause()
                            webView.stopLoading()
                            webView.loadUrl("about:blank")
                            webView.clearHistory()
                            webView.removeAllViews()
                            webView.destroy()
                        } catch (e: Exception) {
                            Log.w("StreamExtractor", "Cancellation cleanup failed: ${e.message}")
                        }
                    }
                }
            }
        }
    } ?: run {
        Log.e("StreamExtractor", "Extraction TIMEOUT for: $embedUrl")
        null
    }
}

@Composable
fun StrumyczekApp(viewModel: MainViewModel = viewModel()) {
    val focusManager = LocalFocusManager.current
    val scope = rememberCoroutineScope()
    val currentTab by viewModel.currentTab.collectAsState()

    // --- GHOST STATE PROTECTION GATE ---
    val tabMoviesState by viewModel.displayedMovies.collectAsState()
    val displayedMovies = if (tabMoviesState.tab == currentTab) tabMoviesState.items else emptyList()

    val recentMovies by viewModel.recentMovies.collectAsState()
    val favoriteMovies by viewModel.favoriteMovies.collectAsState()
    val searchQuery by viewModel.searchQuery.collectAsState()
    val isLoading by viewModel.isLoading.collectAsState()
    val isPaginating by viewModel.isPaginating.collectAsState()
    val errorState by viewModel.errorState.collectAsState()
    val nextEpisodeItem by viewModel.nextEpisodeItem.collectAsState()

    val needsLogin by viewModel.needsLogin.collectAsState()
    val availableSources by viewModel.availableSources.collectAsState()
    val activeMovie by viewModel.activeMovie.collectAsState()
    val selectedSeries by viewModel.selectedSeries.collectAsState()

    val filterState by viewModel.filterState.collectAsState()
    val availableGenres by viewModel.availableGenres.collectAsState()

    var currentScreen by remember { mutableStateOf("home") }
    var streamUrl by remember { mutableStateOf("") }
    var originalUrl by remember { mutableStateOf("") }
    var isWebPlayer by remember { mutableStateOf(false) }
    var loadingMovie by remember { mutableStateOf<MovieItem?>(null) }

    var showSeriesDialog by remember { mutableStateOf(false) }

    var lastBackPressTime by remember { mutableLongStateOf(0L) }
    val context = LocalContext.current

    val topBarFocusRequester = remember { FocusRequester() }
    val contentFocusRequester = remember { FocusRequester() }
    val homeTabFocusRequester = remember { FocusRequester() }

    LaunchedEffect(Unit) {
        delay(100)
        topBarFocusRequester.safeRequestFocus()
    }

    LaunchedEffect(currentScreen, needsLogin, showSeriesDialog) {
        if (currentScreen == "home" && !needsLogin && !showSeriesDialog) {
            delay(300)
            contentFocusRequester.safeRequestFocus()
        }
    }

    BackHandler(enabled = true) {
        when {
            availableSources.isNotEmpty() -> {
                viewModel.closeSourceDialog()
                if (!showSeriesDialog) {
                    scope.launch {
                        delay(50)
                        contentFocusRequester.safeRequestFocus()
                    }
                }
            }
            currentScreen == "player" -> {
                currentScreen = "home"
                viewModel.refreshHistory()
                
                // Wymuszamy zamknięcie ekranów pośrednich
                viewModel.closeSourceDialog()
                showSeriesDialog = false
                viewModel.clearSelectedSeries()
                
                // Opóźnienie pozwala na usunięcie nakładek z drzewa Compose przed rzuceniem fokusu na Grid
                scope.launch {
                    delay(100)
                    contentFocusRequester.safeRequestFocus()
                }
            }
            showSeriesDialog -> {
                showSeriesDialog = false
                viewModel.clearSelectedSeries()
                contentFocusRequester.safeRequestFocus()
            }
            currentScreen == "loading" -> {
                currentScreen = "home"
                loadingMovie = null
                homeTabFocusRequester.safeRequestFocus()
            }
            currentScreen == "home" -> {
                if (currentTab != AppTab.HOME) {
                    viewModel.onTabSelected(AppTab.HOME)
                    homeTabFocusRequester.safeRequestFocus()
                } else {
                    val currentTime = System.currentTimeMillis()
                    if (currentTime - lastBackPressTime < 2000) {
                        (context as? Activity)?.finish()
                    } else {
                        lastBackPressTime = currentTime
                        Toast.makeText(context, "Aby zamknąć naciśnij wstecz jeszcze raz", Toast.LENGTH_SHORT).show()
                    }
                }
            }
        }
    }

    LaunchedEffect(Unit) {
        viewModel.loadDataForTab(AppTab.HOME)
        viewModel.streamEvent.collect { event ->
            when (event) {
                is StreamEvent.Success -> {
                    streamUrl = event.streamUrl
                    originalUrl = event.originalUrl
                    isWebPlayer = event.isWebSource
                    currentScreen = "player"
                    loadingMovie = null
                }
                is StreamEvent.Error -> {
                    Toast.makeText(context, event.message, Toast.LENGTH_LONG).show()
                    currentScreen = "home"
                    loadingMovie = null
                }
            }
        }
    }

    val backDispatcher = LocalOnBackPressedDispatcherOwner.current?.onBackPressedDispatcher

    Surface(
        modifier = Modifier
            .fillMaxSize()
            .onPreviewKeyEvent { event ->
                if (event.key == Key.Back || event.key == Key.Escape) {
                    if (event.type == KeyEventType.KeyUp) {
                        backDispatcher?.onBackPressed()
                    }
                    return@onPreviewKeyEvent true // Konsumujemy zdarzenie natychmiast!
                }
                false
            },
        color = BackgroundDark
    ) {
        Box(modifier = Modifier.fillMaxSize()) {
            if (needsLogin) {
                LoginScreen(
                    onLoginSuccess = { cookies -> viewModel.saveCookies(cookies) },
                    onBack = { (context as? Activity)?.finish() }
                )
            } else {
                // WARSTWA 1: Zawsze w drzewie (Home, Katalog, Biblioteka, Dialogi)
                // Zachowuje pełny stan scrolla i fokusu, ukrywana przez alpha gdy player jest aktywny
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .alpha(if (currentScreen == "home") 1f else 0f)
                        .focusProperties { canFocus = currentScreen == "home" } // Odcina fokus, gdy ekran jest ukryty
                ) {
                    if (errorState != null) {
                        ErrorView(message = errorState!!) { viewModel.loadDataForTab(currentTab) }
                    } else if (isLoading && displayedMovies.isEmpty() && !showSeriesDialog && availableSources.isEmpty() && currentTab == AppTab.HOME) {
                        LoadingView()
                    } else {
                        MainContentWithNavigation(
                            currentTab = currentTab,
                            movies = displayedMovies,
                            recentMovies = recentMovies,
                            favoriteMovies = favoriteMovies,
                            searchQuery = searchQuery,
                            onTabSelected = { viewModel.onTabSelected(it) },
                            onSearchQueryChanged = { viewModel.onSearchQueryChanged(it) },
                            onPlayMovie = { item ->
                                if (item.mediaType == "series") {
                                    viewModel.openSeriesDetails(item)
                                    showSeriesDialog = true
                                } else {
                                    viewModel.prepareMoviePlayback(item)
                                }
                            },
                            onLoadDetails = { viewModel.loadMovieDetails(it) },
                            genres = availableGenres,
                            filterState = filterState,
                            onGenreSelected = { viewModel.onGenreSelected(it) },
                            onSortSelected = { viewModel.onSortSelected(it) },
                            onYearSelected = { viewModel.onYearSelected(it) },
                            isLoading = isLoading,
                            isPaginating = isPaginating,
                            viewModel = viewModel,
                            topBarFocusRequester = topBarFocusRequester,
                            homeTabFocusRequester = homeTabFocusRequester,
                            contentFocusRequester = contentFocusRequester
                        )
                    }

                    if (showSeriesDialog && selectedSeries != null) {
                        SeriesDetailsSheet(
                            seriesLeader = selectedSeries!!,
                            viewModel = viewModel,
                            isFavorite = favoriteMovies.any { it.originalUrl == selectedSeries?.originalUrl },
                            onToggleFavorite = { selectedSeries?.let { viewModel.toggleFavorite(it) } },
                            onEpisodeClick = { episode ->
                                viewModel.prepareMoviePlayback(episode)
                            },
                            onClose = {
                                showSeriesDialog = false
                                viewModel.clearSelectedSeries()
                                contentFocusRequester.safeRequestFocus()
                            },
                            isPlayerActive = (currentScreen == "player")
                        )
                    }

                    if (availableSources.isNotEmpty()) {
                        SourceSelectionScreen(
                            movie = activeMovie ?: MovieItem(
                                originalUrl = "",
                                title = "Wybierz źródło",
                                year = null,
                                thumbnail = null,
                                overview = null,
                                mediaType = null,
                                backdropUrl = null,
                                genres = emptyList(),
                                progress = 0f
                            ),
                            sources = availableSources,
                            isFavorite = favoriteMovies.any { it.originalUrl == activeMovie?.originalUrl },
                            onToggleFavorite = { activeMovie?.let { viewModel.toggleFavorite(it) } },
                            onSourceSelected = { source ->
                                val movieToPlay = activeMovie
                                loadingMovie = movieToPlay
                                currentScreen = "loading"
                                viewModel.playSelectedSource(source)
                            },
                            onBack = {
                                viewModel.closeSourceDialog()
                                if (!showSeriesDialog) {
                                    contentFocusRequester.safeRequestFocus()
                                }
                            }
                        )
                    }
                }

                // WARSTWA 2: Ekran ładowania (Nakładka)
                if (currentScreen == "loading" && loadingMovie != null) {
                    StreamLoadingScreen(movie = loadingMovie!!)
                }

                // WARSTWA 3: Player wideo (Na samej górze)
                if (currentScreen == "player") {
                    PlayerScreen(
                        streamUrl = streamUrl,
                        originalUrl = originalUrl,
                        isWebSource = isWebPlayer,
                        currentMovie = activeMovie,
                        nextEpisode = nextEpisodeItem,
                        isFavorite = favoriteMovies.any { it.originalUrl == activeMovie?.originalUrl },
                        onToggleFavorite = { activeMovie?.let { viewModel.toggleFavorite(it) } },
                        onPreloadNext = { nextEpisodeItem?.let { viewModel.preloadNextEpisodeStream(it) } },
                        onPlayNext = { 
                            val handled = viewModel.playNextEpisode(
                                onLoadingStart = { movie ->
                                    loadingMovie = movie
                                    currentScreen = "loading"
                                }
                            )
                            if (!handled) currentScreen = "home"
                        },
                        onSaveProgress = { pos, dur ->
                            activeMovie?.let { viewModel.savePlaybackProgress(it, pos, dur, isWebPlayer) }
                        },
                        onBackPressed = {
                            currentScreen = "home"
                            viewModel.refreshHistory()
                            
                            // Wymuszamy zamknięcie ekranów pośrednich
                            viewModel.closeSourceDialog()
                            showSeriesDialog = false
                            viewModel.clearSelectedSeries()
                            
                            scope.launch {
                                delay(100)
                                contentFocusRequester.safeRequestFocus()
                            }
                        }
                    )
                }
            }
        }
    }
}

/**
 * MainContentWithNavigation - Główny kontener treści z wymuszoną przebudową stanu przy zmianie sesji zakładki.
 */
@Composable
fun MainContentWithNavigation(
    currentTab: AppTab,
    movies: List<MovieItem>,
    recentMovies: List<MovieItem>,
    favoriteMovies: List<MovieItem>,
    searchQuery: String,
    onTabSelected: (AppTab) -> Unit,
    onSearchQueryChanged: (String) -> Unit,
    onPlayMovie: (MovieItem) -> Unit,
    onLoadDetails: (MovieItem) -> Unit,
    genres: List<String>,
    filterState: CatalogFilterState,
    onGenreSelected: (String?) -> Unit,
    onSortSelected: (String?) -> Unit,
    onYearSelected: (String?) -> Unit,
    isLoading: Boolean,
    isPaginating: Boolean,
    viewModel: MainViewModel,
    topBarFocusRequester: FocusRequester,
    homeTabFocusRequester: FocusRequester,
    contentFocusRequester: FocusRequester
) {
    val tabSessionIds by viewModel.tabSessionIds.collectAsState()
    val currentSessionId = tabSessionIds[currentTab] ?: 0

    Box(modifier = Modifier.fillMaxSize()) {
        Box(
            modifier = Modifier
                .fillMaxSize()
                .focusProperties {
                    exit = { direction ->
                        if (direction == androidx.compose.ui.focus.FocusDirection.Up) topBarFocusRequester else androidx.compose.ui.focus.FocusRequester.Default
                    }
                }
        ) {
            key(currentTab, currentSessionId) {
                when (currentTab) {
                    AppTab.SEARCH -> {
                        SearchContent(
                            searchQuery = searchQuery,
                            onSearchQueryChanged = onSearchQueryChanged,
                            onVoiceSearch = { viewModel.onVoiceSearch(it) },
                            movies = movies,
                            isLoading = isLoading,
                            onMovieClicked = onPlayMovie,
                            contentFocusRequester = contentFocusRequester
                        )
                    }
                    AppTab.HOME -> {
                        OneScreenCinema(
                            viewModel = viewModel,
                            movies = movies,
                            recentMovies = recentMovies,
                            onPlayMovie = onPlayMovie,
                            onMovieFocusedDetails = onLoadDetails,
                            topBarRequester = topBarFocusRequester,
                            contentStartRequester = contentFocusRequester,
                            isHomeTab = true
                        )
                    }
                    AppTab.LIBRARY -> {
                        LibraryScreen(
                            history = recentMovies,
                            favorites = favoriteMovies,
                            onMovieClick = onPlayMovie,
                            contentFocusRequester = contentFocusRequester,
                            topBarRequester = topBarFocusRequester

                    }

                        CatalogScreen(
                            viewModel = viewModel,
                            movies = movies,
                            genres = genres,
                            filterState = filterState,
                            onGenreSelected = onGenreSelected,
                            onSortSelected = onSortSelected,
                            onYearSelected = onYearSelected,
                            onMovieClicked = onPlayMovie,
                            onLoadMore = { viewModel.loadNextPage() },
                            contentFocusRequester = contentFocusRequester,
                            isLoading = isLoading,
                            isPaginating = isPaginating,
                            topBarRequester = topBarFocusRequester
                        )
                    }
                }
            }
        }

        Box(
            modifier = Modifier
                .fillMaxWidth()
                .height(140.dp)
                .background(
                    Brush.verticalGradient(
                        0.0f to Color.Black.copy(alpha = 0.95f),
                        0.4f to Color.Black.copy(alpha = 0.70f),
                        0.7f to Color.Black.copy(alpha = 0.30f),
                        1.0f to Color.Transparent
                    )
                )
                .align(Alignment.TopCenter)
                .zIndex(10f)
        ) {
            TopNavigationBar(
                currentTab = currentTab,
                onTabSelected = onTabSelected,
                focusRequester = topBarFocusRequester,
                homeTabFocusRequester = homeTabFocusRequester,
                contentFocusRequester = contentFocusRequester,
                viewModel = viewModel
            )
        }
    }

    /**
     * displayedMovies powiązane ze stanem TabMoviesState.
     * Zapobiega mignięciu starych danych poprzez jawne dołączenie informacji o zakładce do wyniku.
     */
    val displayedMovies: StateFlow<TabMoviesState> = combine(
        _allRawItems,
        _currentTab,
        _searchQuery,
        _filterState
    ) { rawItems, tab, _, _ ->
        withContext(Dispatchers.Default) {
            val enrichedItems = enrichMoviesWithProgress(rawItems)
                .distinctBy { it.originalUrl }

            val filteredByTab = when (tab) {
                AppTab.HOME -> enrichedItems
                AppTab.MOVIES -> enrichedItems.filter { it.mediaType == "movie" || it.mediaType == null }
                AppTab.SERIES -> enrichedItems.filter { it.mediaType == "series" }
                AppTab.SEARCH -> enrichedItems
                AppTab.LIBRARY -> enrichedItems
            }

            TabMoviesState(tab, groupItems(filteredByTab))
        }
    }.stateIn(
        viewModelScope,
        SharingStarted.WhileSubscribed(5000),
        TabMoviesState(AppTab.HOME, emptyList())
    )

    init {
        val cookies = prefs.getString("cookies", null)
        if (cookies == null) {
            _needsLogin.value = true
            _isLoading.value = false
        } else {
            loadDataForTab(AppTab.HOME)
            startSessionHeartbeat()
        }
    }

    private fun startSessionHeartbeat() {
        heartbeatJob = viewModelScope.launch {
            while (true) {
                delay(900_000L) // 15 minut
                if (prefs.getString("cookies", null) == null) continue

                try {
                    Log.d("Heartbeat", "Pinging Filman.cc to keep session alive...")
                    scraper.getHomePage()
                } catch (e: Exception) {
                    Log.w("Heartbeat", "Ping failed: ${e.message}")
                }
            }
        }
    }

    private fun updateFilter(filterChange: CatalogFilterState.() -> CatalogFilterState) {
        _filterState.value = _filterState.value.filterChange()
        tabCache.remove(_currentTab.value)
        loadDataForTab(_currentTab.value)
    }

    fun onGenreSelected(genre: String?) = updateFilter { copy(genre = genre) }

    fun onSortSelected(sortName: String?) {
        val sortKey = CategoryRegistry.sortOptions[sortName] ?: "sort:date/"
        updateFilter { copy(sort = sortKey) }
    }
    fun onYearSelected(year: String?) = updateFilter { copy(year = year) }

    fun loadDataForTab(tab: AppTab) {
        viewModelScope.launch {
            if (tab == AppTab.LIBRARY) {
                _isLoading.value = true
                refreshHistory()
                refreshFavorites()
                _isLoading.value = false
                return@launch
            }

            val filter = _filterState.value
            val hasFilters = filter.genre != null || filter.year != null || filter.sort != "sort:date/"

            // Sprawdzenie Cache najpierw, aby uniknąć mignięcia
            if (!hasFilters && tabCache.containsKey(tab)) {
                val cached = tabCache[tab]!!
                _allRawItems.value = cached.items
                currentPage = cached.page
                isLastPage = cached.isLastPage
                if (tab == AppTab.HOME) refreshHistory()
                _isLoading.value = false
                return@launch
            }

            // Dopiero gdy nie ma danych w cache lub są filtry, czyścimy i pokazujemy loading
            _isLoading.value = true
            _errorState.value = null
            _needsLogin.value = false
            currentPage = 1
            isLastPage = false
            _allRawItems.value = emptyList()

            try {
                val catId = CategoryRegistry.movieCategories[filter.genre]
                val minLoad = launch { delay(400) }

                val items = when(tab) {
                    AppTab.HOME -> scraper.getHomePage()
                    AppTab.MOVIES -> scraper.getMovies(1, catId, filter.sort, filter.year)
                    AppTab.SERIES -> scraper.getSeries(1, catId, filter.sort, filter.year)
                    else -> emptyList()
                }

                minLoad.join()
                _allRawItems.value = items

                if (!hasFilters) {
                    tabCache[tab] = TabData(items, currentPage, isLastPage)
                }

                if (tab == AppTab.HOME) refreshHistory()
            } catch (e: Exception) {
                handleScraperError(e)
            } finally {
                _isLoading.value = false
            }
        }
    }

    private fun handleScraperError(e: Exception) {
        when (e) {
            is ScraperException.AuthException -> {
                // Czyszczenie wadliwych ciasteczek po stronie lokalnej (aby umożliwić nowe zalogowanie)
                prefs.edit().remove("cookies").apply()
                _needsLogin.value = true
            }
            is ScraperException -> _errorState.value = e.message
            else -> _errorState.value = "Błąd połączenia: ${e.localizedMessage ?: "Nieznany błąd"}"
        }
    }

    // --- FOCUS MATRIX LOGIC ---
    fun updateFocusPosition(tab: AppTab, x: Int, y: Int) {
        val current = _tabFocusMatrix.value.toMutableMap()
        current[tab] = FocusCoordinate(x, y)
        _tabFocusMatrix.value = current
    }

    fun handleBackPress(): Boolean {
        val tab = _currentTab.value
        val currentPos = _tabFocusMatrix.value[tab] ?: FocusCoordinate(0, 0)
        val y = currentPos.y

        return when {
            y >= 2 -> {
                val nextPos = FocusCoordinate(0, 1)
                updateFocusPosition(tab, nextPos.x, nextPos.y)
                viewModelScope.launch { _focusRestoreSignal.emit(nextPos) }
                true
            }
            y == 1 -> {
                val nextPos = FocusCoordinate(currentPos.x, 0)
                updateFocusPosition(tab, nextPos.x, nextPos.y)
                viewModelScope.launch { _focusRestoreSignal.emit(nextPos) }
                true
            }
            else -> false
        }
    }

    fun onMovieFocused(category: String, index: Int, movie: MovieItem? = null) {
        updateFocusState(category, index)
        _lastFocusedCategory.value = category
        val yCoord = if (_currentTab.value == AppTab.HOME) {
            if (category == "Hero") 1 else 2
        } else 2
        updateFocusPosition(_currentTab.value, index, yCoord)

        movie?.let { item ->
            detailDebounceJob?.cancel()
            detailDebounceJob = viewModelScope.launch {
                delay(400)
                loadMovieDetails(item)
            }
        }
    }

    fun resetTabFocus(tab: AppTab) {
        val current = _tabFocusMatrix.value.toMutableMap()
        current.remove(tab)
        _tabFocusMatrix.value = current
        _focusStateMap.value = emptyMap()
        _lastFocusedCategory.value = null

        val resetPos = when(tab) {
            AppTab.HOME -> FocusCoordinate(0, 1)
            else -> FocusCoordinate(0, 0)
        }
        viewModelScope.launch { _focusRestoreSignal.emit(resetPos) }
    }

    fun shouldRestoreFocus(category: String): Boolean = _lastFocusedCategory.value == category

    fun updateFocusState(category: String, index: Int) {
        val currentMap = _focusStateMap.value.toMutableMap()
        currentMap[category] = index
        _focusStateMap.value = currentMap
    }

    fun getFocusIndex(category: String): Int = _focusStateMap.value[category] ?: 0

    fun clearFocusState() {
        _focusStateMap.value = emptyMap()
        _lastFocusedCategory.value = null
        _tabFocusMatrix.value = emptyMap()
    }

    private fun enrichMoviesWithProgress(items: List<MovieItem>): List<MovieItem> {
        return items.map { item ->
            val progress = playbackManager.getProgressPercentage(item.originalUrl)
            item.copy(progress = progress)
        }
    }

    private fun groupItems(items: List<MovieItem>): List<MovieItem> {
        val groupedList = mutableListOf<MovieItem>()
        val tempMap = mutableMapOf<String, MutableList<MovieItem>>()

        items.forEach { item ->
            val match = extractionRegex.find(item.title)
            val isSeriesType = item.mediaType == "series"

            if (match != null || isSeriesType) {
                val cleanTitle = match?.groupValues?.get(3)?.trim() ?: item.title
                if (!tempMap.containsKey(cleanTitle)) {
                    tempMap[cleanTitle] = mutableListOf()
                    groupedList.add(item.copy(title = cleanTitle, mediaType = "series"))
                }
                tempMap[cleanTitle]?.add(item)
            } else {
                groupedList.add(item)
            }
        }

        val finalGroupedList = groupedList.map { leader ->
            val episodes = tempMap[leader.title]
            if (episodes != null) {
                val maxProgress = episodes.maxOfOrNull { it.progress } ?: 0f
                leader.copy(progress = maxProgress)
            } else leader
        }

        _seriesEpisodesMap.value = tempMap
        return finalGroupedList
    }

    fun loadSeriesEpisodes(series: MovieItem) {
        viewModelScope.launch {
            _isLoading.value = true
            try {
                val episodes = scraper.getSeriesEpisodes(series.originalUrl)

                val enrichedEpisodes = episodes.map {
                    it.copy(progress = playbackManager.getProgressPercentage(it.originalUrl))
                }
                _selectedSeriesEpisodes.value = enrichedEpisodes
            } catch (e: Exception) {
                Log.e("MainViewModel", "Błąd ładowania odcinków", e)
            } finally {
                _isLoading.value = false
            }
        }
    }

        _selectedSeries.value = movie

        loadSeriesEpisodes(movie)
    }

    fun clearSelectedSeries() {
        _selectedSeries.value = null
        _selectedSeriesEpisodes.value = emptyList()
        _nextEpisodeItem.value = null
    }

    fun loadMovieDetails(movie: MovieItem) {
        if (movie.overview != null) return
        viewModelScope.launch {
            try {
                val details = scraper.getMovieDetails(movie.originalUrl)

                _allRawItems.value = _allRawItems.value.map { item ->
                    if (item.originalUrl == movie.originalUrl) {
                        item.copy(
                            overview = details.description ?: "Brak opisu.",
                            backdropUrl = details.backdropUrl ?: item.backdropUrl
                        )
                    } else {
                        item
                    }
                }

                if (_selectedSeries.value?.originalUrl == movie.originalUrl) {
                    _selectedSeries.value = _selectedSeries.value?.copy(
                        overview = details.description ?: "Brak opisu.",
                        backdropUrl = details.backdropUrl ?: movie.backdropUrl
                    )
                }

                val filter = _filterState.value
                val hasFilters = filter.genre != null || filter.year != null || filter.sort != "sort:date/"
                if (!hasFilters && tabCache.containsKey(_currentTab.value)) {
                    val currentCache = tabCache[_currentTab.value]!!
                    tabCache[_currentTab.value] = currentCache.copy(items = _allRawItems.value)
                }
            } catch (e: Exception) {
                Log.e("MainViewModel", "Błąd dociągania opisu: ${e.message}")
            }
        }
    }

    fun loadNextPage() {
        if (_isLoading.value || _isPaginating.value || isLastPage || _currentTab.value == AppTab.HOME || _currentTab.value == AppTab.LIBRARY) return

        viewModelScope.launch {
            _isPaginating.value = true
            try {
                val nextPage = currentPage + 1
                val filter = _filterState.value
                val catId = CategoryRegistry.movieCategories[filter.genre]

                val newItems = when(_currentTab.value) {
                    AppTab.MOVIES -> scraper.getMovies(nextPage, catId, filter.sort, filter.year)
                    AppTab.SERIES -> scraper.getSeries(nextPage, catId, filter.sort, filter.year)
                    else -> emptyList()
                }

                if (newItems.isNotEmpty()) {
                    _allRawItems.value = _allRawItems.value + newItems
                    currentPage = nextPage
                    val hasFilters = filter.genre != null || filter.year != null || filter.sort != "sort:date/"
                    if (!hasFilters) {
                        tabCache[_currentTab.value] = TabData(
                            items = _allRawItems.value,
                            page = currentPage,
                            isLastPage = isLastPage
                        )
                    }
                } else {
                    isLastPage = true
                }
            } catch (e: Exception) {
                Log.e("MainViewModel", "Pagination error: ${e.message}")
            } finally {
                _isPaginating.value = false
            }
        }
    }

    fun saveCookies(cookies: String) {
        viewModelScope.launch {
            prefs.edit().putString("cookies", cookies).apply()
            _needsLogin.value = false
            tabCache.clear()
            clearFocusState()
            heartbeatJob?.cancel()
            startSessionHeartbeat()
            
            if (_currentTab.value == AppTab.SEARCH) {
                _allRawItems.value = emptyList()
                val query = _searchQuery.value
                if (query.length >= 2) {
                    _isLoading.value = true
                    performSearch(query)
                }
            } else {
                loadDataForTab(_currentTab.value)
            }
        }
    }

    fun refreshHistory() {
        _recentMovies.value = playbackManager.getHistoryItems()
    }

    fun refreshFavorites() {
        _favoriteMovies.value = favoritesManager.getFavorites()
    }

        if (favoritesManager.isFavorite(movie.originalUrl)) {
            favoritesManager.removeFavorite(movie.originalUrl)
        } else {
            favoritesManager.addFavorite(movie)
        }
        refreshFavorites()
    }

    fun isFavorite(url: String): Boolean = favoritesManager.isFavorite(url)

    /**
     * Zaktualizowana obsługa zmiany zakładki:
     * Zapobieganie agresywnemu czyszczeniu ekranu przed sprawdzeniem cache'u.
     * Dodano mechanizm śledzenia sesji UI dla wymuszenia startu od góry.
     * NAPRAWIONO: Ghost State poprzez synchroniczne zarządzenie _allRawItems przed zmianą _currentTab.
     */
    fun onTabSelected(tab: AppTab) {
        if (_currentTab.value == tab) return

        // --- GHOST STATE STRUCTURAL FIX ---
        // Synchronicznie przygotowujemy listę przedmiotów, aby operator 'combine' w displayedMovies
        // miał natychmiastowy dostęp do danych cache lub pustej listy przed emisją nowego tab.
        if (tab != AppTab.SEARCH && tabCache.containsKey(tab)) {
            _allRawItems.value = tabCache[tab]!!.items
        } else {
            _allRawItems.value = emptyList()
        }

        // Wymuszenie nowej sesji UI dla zakładki (resetuje scrolla w Compose)
        val currentIds = _tabSessionIds.value.toMutableMap()
        currentIds[tab] = (currentIds[tab] ?: 0) + 1
        _tabSessionIds.value = currentIds

        // Czyszczenie starego stanu fokusu, aby zapobiec auto-scrollowaniu do ostatniego elementu
        _focusStateMap.value = emptyMap()
        _lastFocusedCategory.value = null

        _currentTab.value = tab
        _filterState.value = CatalogFilterState()

        if (tab != AppTab.SEARCH) {
            _searchQuery.value = ""
            loadDataForTab(tab)
        } else {
            _isLoading.value = false
        }

        if (tab == AppTab.HOME || tab == AppTab.LIBRARY) {
            refreshHistory()
            refreshFavorites()
        }
    }

    fun onSearchQueryChanged(query: String) {
        _searchQuery.value = query
        if (_currentTab.value == AppTab.SEARCH) {
            searchJob?.cancel()

            // Włączamy spinner natychmiast po wpisaniu znaku (jeśli fraza ma min 2 znaki)
            if (query.length >= 2) {
                _isLoading.value = true
            } else {
                _allRawItems.value = emptyList()
                _isLoading.value = false
            }

            searchJob = viewModelScope.launch {
                delay(800) // Debounce
                performSearch(query)
            }
        }
    }

    fun onVoiceSearch(query: String) {
        _searchQuery.value = query
        if (_currentTab.value == AppTab.SEARCH) {
            searchJob?.cancel()
            _isLoading.value = true // Dodane: włączenie spinnera
            searchJob = viewModelScope.launch {
                performSearch(query)
            }
        }
    }

    private suspend fun performSearch(query: String) {
        if (query.length >= 2) {
            try {
                val results = scraper.search(query)
                _allRawItems.value = results

                // Pobieramy URL pierwszego elementu jako unikalny stempel nowej paczki danych
                val firstResultUrl = results.firstOrNull()?.originalUrl

                // Czekamy (maks 8 sekund) aż konkretny, nowy element pojawi się w displayedMovies.
                // To deterministyczna weryfikacja, że dane napłynęły przez combine do warstwy UI.
                withTimeoutOrNull(8000) {
                    displayedMovies.first { state ->
                        state.tab != AppTab.SEARCH ||
                                (results.isEmpty() && state.items.isEmpty()) ||
                                (firstResultUrl != null && state.items.any { it.originalUrl == firstResultUrl })
                    }
                }

            } catch (e: Exception) {
                Log.e("MainViewModel", "Search error", e)
                handleScraperError(e) // ZMIANA: Przekazanie wyjątku do handlera
            } finally {
                _isLoading.value = false // Wyłączamy spinner DOPIERO, gdy nowe okładki są gotowe
            }
        } else {
            _allRawItems.value = emptyList()
            _isLoading.value = false
        }
    }

    fun getSeasonsFor(seriesTitle: String): Map<Int, List<EpisodeInfo>> {
        val episodes = _selectedSeriesEpisodes.value
        if (episodes.isEmpty()) return emptyMap()
        return episodes.map { item ->
            val match = extractionRegex.find(item.title)
            if (match != null) {
                EpisodeInfo(
                    season = match.groupValues[1].toIntOrNull() ?: 1,
                    episode = match.groupValues[2].toIntOrNull() ?: 1,
                    cleanTitle = seriesTitle,
                    fullMovieItem = item
                )
            } else {
                EpisodeInfo(1, 0, seriesTitle, item)
            }
        }
            .sortedWith(compareBy({ it.season }, { it.episode }))
            .groupBy { it.season }
    }

    private fun getChronologicalEpisodes(): List<MovieItem> {
        return getSeasonsFor("").values.flatten().map { it.fullMovieItem }
    }

    private fun determineNextEpisode(currentMovie: MovieItem) {
        val chronological = getChronologicalEpisodes()
        if (chronological.isNotEmpty()) {
            val currentIndex = chronological.indexOfFirst { it.originalUrl == currentMovie.originalUrl }
            if (currentIndex != -1 && currentIndex + 1 < chronological.size) {
                _nextEpisodeItem.value = chronological[currentIndex + 1]
            } else {
                _nextEpisodeItem.value = createPlaybackEndItem()
            }
        } else if (currentMovie.mediaType == "episode") {
            // Context recovery for History/Library playback
            viewModelScope.launch {
                try {
                    val match = extractionRegex.find(currentMovie.title)
                    val cleanTitle = match?.groupValues?.get(3)?.trim() ?: currentMovie.title
                    val results = scraper.search(cleanTitle)
                    val series = results.find { it.mediaType == "series" }

                    if (series != null) {
                        val seriesEpisodes = scraper.getSeriesEpisodes(series.originalUrl)
                        val enriched = seriesEpisodes.map {
                            it.copy(progress = playbackManager.getProgressPercentage(it.originalUrl))
                        }
                        _selectedSeriesEpisodes.value = enriched

                        val recoveredChronological = getChronologicalEpisodes()
                        val idx = recoveredChronological.indexOfFirst { it.originalUrl == currentMovie.originalUrl }
                        if (idx != -1 && idx + 1 < recoveredChronological.size) {
                            _nextEpisodeItem.value = recoveredChronological[idx + 1]
                        } else {
                            _nextEpisodeItem.value = createPlaybackEndItem()
                        }
                    } else {
                        _nextEpisodeItem.value = createPlaybackEndItem()
                    }
                } catch (e: Exception) {
                    _nextEpisodeItem.value = createPlaybackEndItem()
                }
            }
        } else {
            _nextEpisodeItem.value = null
        }
    }

    private fun createPlaybackEndItem() = MovieItem(
        originalUrl = "playback_end",
        title = "Koniec odtwarzania",
        year = null,
        thumbnail = null,
        overview = "Koniec listy. Wróć do menu by wybrać następny.",
        mediaType = "end",
        backdropUrl = null,
        genres = emptyList(),
        progress = 0f
    )

    fun prepareMoviePlayback(movie: MovieItem) {
        determineNextEpisode(movie)
        _preloadedNextStream.value = null
        preloadJob?.cancel()
        viewModelScope.launch {
            _isLoading.value = true
            try {
                val details = scraper.getMovieDetails(movie.originalUrl)
                if (details.links.isEmpty()) {
                    _streamEvent.send(StreamEvent.Error("Brak dostępnych linków"))
                } else {
                    val enrichedMovie = movie.copy(
                        overview = details.description ?: movie.overview,
                        backdropUrl = details.backdropUrl ?: movie.backdropUrl
                    )
                    _activeMovie.value = enrichedMovie
                    _availableSources.value = details.links
                }
            } catch (e: Exception) {
                _streamEvent.send(StreamEvent.Error("Błąd pobierania źródeł"))
            } finally {
                _isLoading.value = false
            }
        }
    }

    fun playSelectedSource(source: StreamItem) {
        val movie = _activeMovie.value ?: return
        val otherSources = _availableSources.value.filter { it.url != source.url }
        val fallbackQueue = listOf(source) + otherSources

        if (_isGeneratingStream.value) return

        viewModelScope.launch {
            _isGeneratingStream.value = true
            var success = false
            try {
                for (currentSource in fallbackQueue) {
                    Log.d("MainViewModel", "Próba ekstrakcji z: ${currentSource.name}")
                    val extractedUrl = streamExtractor.extractVideoUrl(currentSource.url)
                    if (extractedUrl != null) {
                        _streamEvent.send(StreamEvent.Success(extractedUrl, movie.originalUrl, false))
                        success = true
                        break
                    }
                }

                if (!success) {
                    // Ostateczny fallback: Próba otwarcia pierwotnie wybranego źródła w WebPlayerze
                    _streamEvent.send(StreamEvent.Success(source.url, movie.originalUrl, true))
                }
            } catch (e: Exception) {
                Log.e("MainViewModel", "Błąd podczas fallbacku streamów", e)
                _streamEvent.send(StreamEvent.Error("Wystąpił błąd podczas generowania strumienia."))
            } finally {
                _isGeneratingStream.value = false
            }
        }
    }

    fun preloadNextEpisodeStream(episode: MovieItem) {
        if (episode.originalUrl == "playback_end") return
        preloadJob?.cancel()
        preloadJob = viewModelScope.launch {
            try {
                val details = scraper.getMovieDetails(episode.originalUrl)
                if (details.links.isNotEmpty()) {
                    val sortedSources = details.links.sortedByDescending {
                        it.language.contains("Lektor", ignoreCase = true)
                    }

                    for (source in sortedSources) {
                        val extractedUrl = streamExtractor.extractVideoUrl(source.url)
                        if (extractedUrl != null) {
                            _preloadedNextStream.value = StreamEvent.Success(
                                streamUrl = extractedUrl,
                                originalUrl = episode.originalUrl,
                                isWebSource = false
                            )
                            break
                        }
                    }
                }
            } catch (e: Exception) {
                Log.e("MainViewModel", "Preload error", e)
            }
        }
    }

    fun closeSourceDialog() {
        _availableSources.value = emptyList()
        _activeMovie.value = null
        _preloadedNextStream.value = null
        preloadJob?.cancel()
    }

    fun playNextEpisode(onLoadingStart: (MovieItem) -> Unit): Boolean {
        val next = _nextEpisodeItem.value ?: return false
        if (next.originalUrl == "playback_end") return false

        val preloaded = _preloadedNextStream.value
        if (preloaded != null && preloaded.originalUrl == next.originalUrl) {
            viewModelScope.launch {
                // Wymuszenie pokazania ekranu ładowania w celu schowania playera
                _isLoading.value = true 
                _activeMovie.value = next
                determineNextEpisode(next)
                
                // Krótkie opóźnienie dla stabilizacji ExoPlayera przed otrzymaniem nowego URL
                delay(300) 
                
                _streamEvent.send(preloaded)
                _preloadedNextStream.value = null
                _isLoading.value = false
            }
            return true
        } else {
            onLoadingStart(next)
            autoPlayNextEpisode(next)
            return true
        }
    }

    private fun autoPlayNextEpisode(episode: MovieItem) {
        viewModelScope.launch {
            _activeMovie.value = episode
            determineNextEpisode(episode)
            try {
                val details = scraper.getMovieDetails(episode.originalUrl)
                _activeMovie.value = episode.copy(
                    overview = details.description ?: episode.overview,
                    backdropUrl = details.backdropUrl ?: episode.backdropUrl
                )
                _availableSources.value = details.links

                if (details.links.isNotEmpty()) {
                    val sortedSources = details.links.sortedByDescending {
                        it.language.contains("Lektor", ignoreCase = true)
                    }

                    var found = false
                    for (source in sortedSources) {
                        val extractedUrl = streamExtractor.extractVideoUrl(source.url)
                        if (extractedUrl != null) {
                            _streamEvent.send(StreamEvent.Success(extractedUrl, episode.originalUrl, false))
                            found = true
                            break
                        }
                    }

                    if (!found) {
                        // Fallback to first available source in WebView
                        _streamEvent.send(StreamEvent.Success(sortedSources.first().url, episode.originalUrl, true))
                    }
                } else {
                    _streamEvent.send(StreamEvent.Error("Brak dostępnych źródeł dla kolejnego odcinka."))
                }
            } catch (e: Exception) {
                _streamEvent.send(StreamEvent.Error("Błąd automatycznego odtwarzania: ${e.message}"))
            }
        }
    }

    /**
     * Centralna metoda zapisu historii odtwarzania.
     */
    fun savePlaybackProgress(movie: MovieItem, position: Long, duration: Long, isWebSource: Boolean) {
        if (isWebSource || position < 10_000 || duration <= 0) return

        // Zabezpieczenie miniaturki dla odcinka
        val thumbnailToSave = movie.thumbnail ?: _selectedSeries.value?.thumbnail ?: _selectedSeries.value?.backdropUrl

        playbackManager.saveProgress(
            movie.originalUrl,
            movie.title,
            thumbnailToSave,
            movie.mediaType,
            position,
            duration
        )

        // AKTUALIZACJA STANU UI
        if (movie.mediaType == "episode") {
            val updatedEpisodes = _selectedSeriesEpisodes.value.map { ep ->
                if (ep.originalUrl == movie.originalUrl) {
                    ep.copy(progress = playbackManager.getProgressPercentage(ep.originalUrl))
                } else ep
            }
            _selectedSeriesEpisodes.value = updatedEpisodes

            _selectedSeries.value?.let { series ->
                val maxProg = updatedEpisodes.maxOfOrNull { it.progress } ?: 0f
                _selectedSeries.value = series.copy(progress = maxProg)

                playbackManager.saveProgress(
                    series.originalUrl,
                    series.title,
                    series.thumbnail ?: series.backdropUrl,
                    "series",
                    position,
                    duration
                )
            }
        } else {
            _allRawItems.value = _allRawItems.value.map { item ->
                if (item.originalUrl == movie.originalUrl) {
                    item.copy(progress = playbackManager.getProgressPercentage(item.originalUrl))
                } else item
            }
        }

        refreshHistory()
    }

    )

    cardWidth: Dp = 120.dp,

 * dwufazową animacją (skala + opóźnione detale) oraz paskiem postępu.
 */
@Composable
fun EpisodeRowItem(
    episode: EpisodeInfo,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
    fallbackThumbnail: String? = null
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()
    val isPressed by interactionSource.collectIsPressedAsState()
    val isTv = LocalIsTv.current

    val isVisualFocused = (isFocused && isTv) || isPressed

    // Stan dla drugiej fazy animacji (detale)
    var showDetails by remember { mutableStateOf(false) }

    // Faza 1: Płynne skalowanie przy wejściu w fokus
    val scale by animateFloatAsState(
        targetValue = if (isVisualFocused) 1.05f else 1f,
        animationSpec = tween(durationMillis = 250, easing = FastOutSlowInEasing),
        label = "scale"
    )

    // Faza 2: Wyzwalacz opóźnionego wyświetlania opisu
    LaunchedEffect(isFocused) {
        if (isFocused && isTv) {
            delay(1000)
            showDetails = true
        } else if (!isFocused) {
            showDetails = false
        }
    }

    val backgroundColor = if (isVisualFocused) Color.White else Color.Transparent
    val textColor = if (isVisualFocused) Color.Black else TextWhite
    val subTextColor = if (isVisualFocused) Color.DarkGray else TextGrey

    Row(
        modifier = modifier
            .fillMaxWidth()
            .graphicsLayer {
                scaleX = scale
                scaleY = scale
            }
            .clip(RoundedCornerShape(8.dp))
            .background(backgroundColor)
            .then(
                if (isVisualFocused) {
                    Modifier
                        // 1. Zewnętrzna krawędź (AccentMint)
                        .border(width = 1.dp, color = AccentMint, shape = RoundedCornerShape(8.dp))
                        .padding(1.dp)
                        // 2. Środek (Biały)
                        .border(width = 2.dp, color = Color.White, shape = RoundedCornerShape(7.dp))
                        .padding(2.dp)
                        // 3. Wewnętrzna krawędź (AccentMint)
                        .border(width = 1.dp, color = AccentMint, shape = RoundedCornerShape(5.dp))
                } else Modifier
            )
            .clickable(
                interactionSource = interactionSource,
                indication = null
            ) { 
                showDetails = !showDetails
                onClick() 
            }
            .focusable(interactionSource = interactionSource)
            .padding(8.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        // 1. Miniaturka z opcjonalnym paskiem postępu
        Box(
            modifier = Modifier
                .width(140.dp)
                .aspectRatio(16f / 9f)
                .clip(RoundedCornerShape(4.dp))
                .background(Color.Black)
        ) {
            AsyncImage(
                model = ImageRequest.Builder(LocalContext.current)
                    .data(episode.fullMovieItem.thumbnail ?: fallbackThumbnail)
                    .crossfade(true)
                    .build(),
                contentDescription = "Miniatura odcinka: ${episode.cleanTitle}",
                // ContentScale.Crop zapewnia, że nawet pionowy plakat serialu (fallback)
                // wypełni całą powierzchnię poziomego prostokąta 16:9
                contentScale = ContentScale.Crop,
                modifier = Modifier.fillMaxSize()
            )

            // Pasek postępu na miniaturce (Cinematic Style)
            if (episode.fullMovieItem.progress > 0) {
                // Tło paska (Track)
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(4.dp)
                        .align(Alignment.BottomStart)
                        .background(Color.White.copy(alpha = 0.2f))
                )
                // Właściwy postęp
                Box(
                    modifier = Modifier
                        .fillMaxWidth(episode.fullMovieItem.progress)
                        .height(4.dp)
                        .align(Alignment.BottomStart)
                        .background(AccentMint)
                )
            }
        }

        Spacer(modifier = Modifier.width(16.dp))

        // 2. Informacje tekstowe
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = "${episode.episode}. ${episode.cleanTitle}",
                color = textColor,
                style = MaterialTheme.typography.titleMedium,
                fontWeight = FontWeight.Bold,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )

            Text(
                text = "S${episode.season}E${episode.episode} • ${episode.fullMovieItem.year ?: ""}",
                color = subTextColor,
                style = MaterialTheme.typography.labelMedium,
                modifier = Modifier.padding(top = 4.dp)
            )

            // Faza 2 Animacji: Opis pojawiający się po 1s fokusu
            AnimatedVisibility(
                visible = showDetails && !episode.fullMovieItem.overview.isNullOrEmpty(),
                enter = fadeIn(),
                exit = fadeOut()
            ) {
                Text(
                    text = episode.fullMovieItem.overview ?: "",
                    color = subTextColor,
                    style = MaterialTheme.typography.bodySmall,
                    maxLines = 2,
                    overflow = TextOverflow.Ellipsis,
                    modifier = Modifier.padding(top = 4.dp)
                )
            }
        }
    }
}

/**
 * FocusAwareLazyRow - Zoptymalizowany wiersz LazyRow dla interfejsu TV.
 * Rozwiązuje problem niepowiązanych FocusRequesterów poprzez re-użycie zewnętrznego requestera dla elementu startowego.
 */
@Composable
fun FocusAwareLazyRow(
    title: String,
    items: List<MovieItem>,
    onItemFocused: (MovieItem, Int) -> Unit,
    onItemClicked: (MovieItem) -> Unit,
    initialIndex: Int,
    modifier: Modifier = Modifier,
    focusRequester: FocusRequester? = null,
    restoreFocus: Boolean = false
) {
    val listState = rememberLazyListState(initialFirstVisibleItemIndex = initialIndex)

    // Re-używamy przekazanego focusRequester dla startowego elementu, aby połączyć drzewo fokusu
    val itemRequesters = remember(items, focusRequester, initialIndex) {
        List(items.size) { index ->
            if (index == initialIndex && focusRequester != null) {
                focusRequester
            } else {
                FocusRequester()
            }
        }
    }

    LaunchedEffect(focusRequester) {
        if (focusRequester != null && !restoreFocus && initialIndex in items.indices) {
            try {
                // To wywołanie jest teraz bezpieczne, bo requester jest podpięty do elementu niżej
                itemRequesters[initialIndex].requestFocus()
            } catch (e: Exception) {
            }
        }
    }

    Column(modifier = modifier.fillMaxWidth()) {
        Text(
            text = title,
            style = MaterialTheme.typography.headlineSmall,
            fontWeight = FontWeight.Bold,
            color = Color(0xFFE0E0E0),
            modifier = Modifier.padding(start = 50.dp, bottom = 20.dp)
        )

        LazyRow(
            state = listState,
            contentPadding = PaddingValues(start = 50.dp, end = 50.dp, bottom = 50.dp),
            horizontalArrangement = Arrangement.spacedBy(20.dp),
            modifier = Modifier.fillMaxWidth()
        ) {
            itemsIndexed(
                items = items,
                key = { _, movie -> "${title}_${movie.originalUrl}" }
            ) { index, movie ->

                LaunchedEffect(restoreFocus) {
                    if (restoreFocus && index == initialIndex) {
                        try {
                            itemRequesters[index].requestFocus()
                        } catch (e: Exception) {}
                    }
                }

                Box(
                    modifier = Modifier.onFocusChanged { focusState ->
                        if (focusState.isFocused) {
                            onItemFocused(movie, index)
                        }
                    }
                ) {
                    CinematicMovieCard(
                        movie = movie,
                        onClick = { onItemClicked(movie) },
                        onLongFocus = { },
                        focusRequester = itemRequesters[index]
                    )
                }
            }
        }
    }
}

/**
 * SeriesDetailsSheet - Arkusz szczegółów serialu z ulepszoną logiką fokusu i nawigacji.
 */
@OptIn(ExperimentalComposeUiApi::class)
@Composable
fun SeriesDetailsSheet(
    seriesLeader: MovieItem,
    viewModel: MainViewModel,
    isFavorite: Boolean,
    onToggleFavorite: () -> Unit,
    onEpisodeClick: (MovieItem) -> Unit,
    onClose: () -> Unit,
    isPlayerActive: Boolean = false
) {
    val focusManager = LocalFocusManager.current
    val selectedEpisodes by viewModel.selectedSeriesEpisodes.collectAsState()
    val availableSources by viewModel.availableSources.collectAsState()
    var lastClickedSource by remember { mutableStateOf<String?>(null) }

    val seasonsMap = remember(seriesLeader, selectedEpisodes) {
        viewModel.getSeasonsFor(seriesLeader.title)
    }

    val seasonsList = remember(seasonsMap) {
        seasonsMap.keys.sorted()
    }

    var selectedSeason by remember { mutableStateOf(seasonsList.firstOrNull() ?: 1) }

    val currentEpisodes = remember(seasonsMap, selectedSeason) {
        seasonsMap[selectedSeason] ?: emptyList()
    }

    val episodeRequesters = remember(currentEpisodes) {
        currentEpisodes.associate { it.fullMovieItem.originalUrl to FocusRequester() }
    }

    val isLoading by viewModel.isLoading.collectAsState()

    // Focus Requesters dla jawnego routingu fokusu
    val backButtonRequester = remember { FocusRequester() }
    val favoriteButtonRequester = remember { FocusRequester() }
    val playButtonRequester = remember { FocusRequester() }
    val seasonListRequester = remember { FocusRequester() }
    val firstEpisodeRequester = remember { FocusRequester() }

    // State dla przewijania lewej kolumny
    val leftPanelScrollState = rememberScrollState()

    // Synchronizacja domyślnego sezonu po załadowaniu danych
    LaunchedEffect(seasonsList) {
        if (seasonsList.isNotEmpty() && !seasonsList.contains(selectedSeason)) {
            selectedSeason = seasonsList.first()
        }
    }

    // Startowy fokus przy otwarciu arkusza
    LaunchedEffect(Unit) {
        playButtonRequester.smartRequestFocus()
    }

    LaunchedEffect(seasonsList) {
        if (seasonsList.isNotEmpty() && !isPlayerActive) {
            try {
                playButtonRequester.requestFocus()
            } catch (e: Exception) {}
        }
    }

    // Restore focus when returning from player
    LaunchedEffect(isPlayerActive) {
        if (!isPlayerActive) {
            delay(400) // Stabilizacja układu

            // ZAAWANSOWANY ROUTING FOKUSU: Przywraca fokus na dokładnie ten element, z którego użytkownik wszedł do playera
            if (lastClickedSource == "MAIN_PLAY") {
                playButtonRequester.smartRequestFocus(maxRetries = 15)
            } else if (lastClickedSource != null) {
                // FIX: Przywraca fokus na dokładnie ten odcinek, z którego użytkownik wszedł do playera
                episodeRequesters[lastClickedSource]?.smartRequestFocus(maxRetries = 15)
                    ?: playButtonRequester.smartRequestFocus(maxRetries = 15)
            } else {
                playButtonRequester.smartRequestFocus(maxRetries = 15)
            }
        }
    }

    LaunchedEffect(availableSources.isEmpty()) {
        if (availableSources.isEmpty() && lastClickedSource != null && !isPlayerActive) {
            delay(150) // Czekamy aż okno źródeł zniknie z drzewa
            try {
                if (lastClickedSource == "MAIN_PLAY") {
                    playButtonRequester.requestFocus()
                } else {
                    episodeRequesters[lastClickedSource]?.requestFocus()
                }
            } catch (e: Exception) {}
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .alpha(if (isPlayerActive) 0f else 1f)
            .background(BackgroundDark)
    ) {
        AsyncImage(
            model = ImageRequest.Builder(LocalContext.current)

            modifier = Modifier
                .fillMaxSize()
                .background(
                    Brush.verticalGradient(
                        colors = listOf(Color.Transparent, BackgroundDark.copy(alpha = 0.9f))
                    )
                )
        )

        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(horizontal = 60.dp, vertical = 40.dp)
        ) {
            // TOP BAR: Nawigacja i Ulubione z jawnym routingiem fokusu

                modifier = Modifier.fillMaxWidth(),
                verticalAlignment = Alignment.CenterVertically
            ) {
                IconButton(
                    onClick = onClose,
                    modifier = Modifier
                        .focusRequester(backButtonRequester)
                        .focusProperties {
                            up = androidx.compose.ui.focus.FocusRequester.Cancel

                            down = playButtonRequester
                        }
                ) {
                    Icon(Icons.Rounded.ArrowBack, contentDescription = "Wróć", tint = Color.White)

                Spacer(modifier = Modifier.width(16.dp))

                Text(
                    text = seriesLeader.title,
                    style = MaterialTheme.typography.headlineLarge,
                    color = TextWhite,
                    fontWeight = FontWeight.Black,
                    modifier = Modifier.weight(1f)
                )

                TvDialogButton(
                    text = if (isFavorite) "USUŃ Z ULUBIONYCH" else "DODAJ DO ULUBIONYCH",
                    icon = if (isFavorite) Icons.Rounded.Favorite else Icons.Rounded.FavoriteBorder,

                            down = if (currentEpisodes.isNotEmpty()) firstEpisodeRequester else playButtonRequester
                        }
                )
            }

            Spacer(modifier = Modifier.height(24.dp))

            // FIX: Scroll state for the episode list
            val episodeListState = androidx.compose.foundation.lazy.rememberLazyListState()

            // FIX: Auto-scroll to top when season changes
            LaunchedEffect(selectedSeason) {
                episodeListState.scrollToItem(0)

            Row(modifier = Modifier.fillMaxSize()) {

                    val flattenedEpisodes = remember(seasonsMap) { seasonsMap.values.flatten() }

                    val episodeToPlay = remember(flattenedEpisodes) {

                        val lastWatchedIndex = flattenedEpisodes.indexOfLast { it.fullMovieItem.progress > 0f }

                        if (lastWatchedIndex != -1) {
                            val lastWatched = flattenedEpisodes[lastWatchedIndex]

                            // 2. Jeśli najdalszy odcinek nie jest w pełni ukończony, kontynuuj go
                            if (lastWatched.fullMovieItem.progress < 1.0f) {

                    targetState = selectedSeason,

                    // Pobieramy odcinki dla animowanego stanu, aby zapobiec mignięciom

@OptIn(ExperimentalComposeUiApi::class)
@Composable
fun SeasonTabItem(
    number: Int,
    isSelected: Boolean,
    onClick: () -> Unit,
    rightRequester: FocusRequester,
    modifier: Modifier = Modifier
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    // FIX: Standardized focus state (solid white) takes priority over selection
    val backgroundColor = when {
        isFocused -> Color.White.copy(alpha = 0.9f)
        isSelected -> AccentMint
        else -> Color.White.copy(alpha = 0.05f)
    }

    val textColor = when {
        isFocused -> Color.Black
        isSelected -> BackgroundDark
        else -> Color.White
    }

    Surface(
        modifier = modifier
            .fillMaxWidth(0.6f)
            .clip(RoundedCornerShape(8.dp))
            .focusProperties {
                right = rightRequester
                left = androidx.compose.ui.focus.FocusRequester.Cancel
            }
            .clickable(interactionSource = interactionSource, indication = null) { onClick() }
            .focusable(interactionSource = interactionSource),
        shape = RoundedCornerShape(8.dp),
        color = backgroundColor // Used dynamic background instead of hardcoded conditional
    ) {
        Text(
            text = "Sezon $number",
            modifier = Modifier.padding(horizontal = 20.dp, vertical = 12.dp),
            color = textColor, // Used dynamic text color
            fontWeight = if (isSelected || isFocused) FontWeight.Bold else FontWeight.Medium,
            fontSize = 18.sp
        )
    }
}

/**
 * Pędzel Shimmer używany do animacji ładowania szkieletowego.
 */
@Composable
fun shimmerBrush(): Brush {
    val transition = rememberInfiniteTransition(label = "shimmer")
    val translateAnim = transition.animateFloat(
        initialValue = 0f,
        targetValue = 1000f,
        animationSpec = infiniteRepeatable(
            animation = tween(durationMillis = 1200, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "shimmerTranslation"
    )

    val colors = listOf(
        Color.White.copy(alpha = 0.03f),
        Color.White.copy(alpha = 0.08f),
        Color.White.copy(alpha = 0.03f)
    )

    return Brush.linearGradient(
        colors = colors,
        start = Offset.Zero,
        end = Offset(x = translateAnim.value, y = translateAnim.value)
    )
}

        sources.groupBy { it.language.ifEmpty { "Oryginał" } }

        mutableStateOf(
            languages.find { it.contains("Lektor", ignoreCase = true) }
                ?: languages.firstOrNull()
                ?: ""
        )

    val currentSources = groupedSources[selectedLanguage] ?: emptyList()

/**
 * TopNavigationBar - Pasek nawigacyjny z logiką inteligentnego fokusu (Smart Down)
 * umożliwiającą powrót do ostatnio aktywnego elementu treści.
 */
@OptIn(ExperimentalComposeUiApi::class)
@Composable
fun TopNavigationBar(
    currentTab: AppTab,
    onTabSelected: (AppTab) -> Unit,
    focusRequester: FocusRequester,
    homeTabFocusRequester: FocusRequester,
    contentFocusRequester: FocusRequester,
    viewModel: MainViewModel
) {
    val tabRequesters = remember { AppTab.values().associateWith { FocusRequester() } }
    var initialFocusDone by remember { mutableStateOf(false) }
    val focusMatrix by viewModel.tabFocusMatrix.collectAsState()

    AppTab.values().forEach { tab ->
        val isSelected = currentTab == tab
        LaunchedEffect(isSelected) {
            if (isSelected && !initialFocusDone) {
                yield()
                delay(50)
                tabRequesters[tab]?.safeRequestFocus()
                initialFocusDone = true
            }
        }
    }

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 50.dp, vertical = 20.dp)
            .focusProperties {
                enter = { tabRequesters[currentTab] ?: FocusRequester.Default }
            },
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.spacedBy(30.dp)
    ) {
        Text(
            "FILMANTV",
            color = AccentMint,
            style = MaterialTheme.typography.headlineMedium,
            fontWeight = FontWeight.Black
        )

        AppTab.values().forEachIndexed { index, tab ->
            val isSelected = currentTab == tab
            val interactionSource = remember { MutableInteractionSource() }
            val isFocused by interactionSource.collectIsFocusedAsState()
            val isPressed by interactionSource.collectIsPressedAsState()
            val isTv = LocalIsTv.current
            
            // Warunek wizualnego fokusu: sfokusowany na TV lub naciśnięty (dotyk)
            val showVisualFocus = (isFocused && isTv) || isPressed

            Box(
                modifier = Modifier
                    .focusRequester(tabRequesters[tab]!!)
                    .then(if (tab == AppTab.HOME) Modifier.focusRequester(homeTabFocusRequester) else Modifier)
                    .then(if (isSelected) Modifier.focusRequester(focusRequester) else Modifier)
                    .onFocusChanged {
                        if (it.isFocused) {
                            // Raportowanie pozycji X w macierzy fokusu (Y=0)
                            viewModel.updateFocusPosition(tab, index, 0)
                        }
                    }
                    .onPreviewKeyEvent { event ->
                        // Inteligentne zejście (Smart Down) do treści po naciśnięciu strzałki w dół
                        if (event.type == KeyEventType.KeyDown && event.key == Key.DirectionDown) {
                            val lastCoord = focusMatrix[currentTab]
                            // Przekierowanie do zakotwiczenia treści (Hero/Filtry/Grid)
                            contentFocusRequester.safeRequestFocus()
                            true
                        } else false
                    }
                    .clickable(
                        interactionSource = interactionSource,
                        indication = null,
                        onClick = { onTabSelected(tab) }
                    )
                    .focusable(interactionSource = interactionSource),
                contentAlignment = Alignment.Center
            ) {
                Column(horizontalAlignment = Alignment.CenterHorizontally) {
                    
                    // Wewnętrzny Box na Tekst/Ikonę - tutaj ląduje tło i padding
                    Box(
                        modifier = Modifier
                            .background(
                                color = if (showVisualFocus) Color.White else Color.Transparent,
                                shape = RoundedCornerShape(16.dp)
                            )
                            .padding(horizontal = 20.dp, vertical = 8.dp),
                        contentAlignment = Alignment.Center
                    ) {
                        if (tab == AppTab.SEARCH) {
                            Icon(
                                imageVector = Icons.Rounded.Search,
                                contentDescription = "Szukaj",
                                tint = if (showVisualFocus) Color.Black else if (isSelected) Color.White else Color.Gray,
                                modifier = Modifier.size(24.dp)
                            )
                        } else {
                            Text(
                                text = tab.label,
                                color = if (showVisualFocus) Color.Black else if (isSelected) Color.White else Color.Gray,
                                fontWeight = if (isSelected) FontWeight.Bold else FontWeight.Normal
                            )
                        }
                    }

                    // Wskaźnik wyboru (Podkreślenie) pod tłem
                    Box(
                        modifier = Modifier
                            .offset(y = (-6).dp)
                            .height(3.dp)
                            .width(24.dp)
                            .background(
                                color = if (isSelected && !showVisualFocus) AccentMint else Color.Transparent,
                                shape = CircleShape
                            )
                    )
                }
            }
        }
    }
}

 * Zapewnia wyraźną informację zwrotną (skala i zmiana kolorów) podczas fokusu.
 */
@Composable
fun TvDialogButton(
    text: String,
    icon: ImageVector,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
    isPrimary: Boolean = true
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()
    val isPressed by interactionSource.collectIsPressedAsState()
    val isTv = LocalIsTv.current

    val isVisualFocused = (isFocused && isTv) || isPressed

    // Animacja skali dla lepszej widoczności fokusu
    val scale by animateFloatAsState(
        targetValue = if (isVisualFocused) 1.05f else 1.0f,
        label = "buttonScale"
    )

    // Dynamiczna logika kolorów dla zapewnienia wysokiego kontrastu
    val backgroundColor = when {
        isVisualFocused -> Color.White.copy(alpha = 0.9f)
        isPrimary -> Color.White.copy(alpha = 0.15f)
        else -> Color.Transparent
    }

    val contentColor = when {
        isVisualFocused -> Color.Black
        isPrimary -> Color.White
        else -> Color.Gray
    }

    Row(
        modifier = modifier
            .graphicsLayer {
                scaleX = scale
                scaleY = scale
            }
            .clickable(
                interactionSource = interactionSource,
                indication = null,
                onClick = onClick
            )
            .background(backgroundColor, RoundedCornerShape(8.dp))
            .border(
                width = 2.dp,
                color = if (isVisualFocused) AccentMint else Color.Transparent,
                shape = RoundedCornerShape(8.dp)
            )
            .padding(horizontal = 20.dp, vertical = 12.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Icon(
            imageVector = icon,
            contentDescription = null,
            tint = contentColor,
            modifier = Modifier.size(20.dp)
        )
        Spacer(modifier = Modifier.width(12.dp))
        Text(
            text = text,
            color = contentColor,
            style = MaterialTheme.typography.bodyLarge,
            fontWeight = FontWeight.Bold
        )
    }

/**
 * CatalogScreen - Zoptymalizowany ekran katalogu z naprawionym mechanizmem paginacji
 * oraz stabilnym drzewem fokusu (brak Crossfade).
 */
@Composable
fun CatalogScreen(
    viewModel: MainViewModel,
    movies: List<MovieItem>,
    genres: List<String>,
    filterState: CatalogFilterState,
    onGenreSelected: (String?) -> Unit,
    onSortSelected: (String?) -> Unit,
    onYearSelected: (String?) -> Unit,
    onMovieClicked: (MovieItem) -> Unit,
    onLoadMore: () -> Unit,
    contentFocusRequester: FocusRequester,
    topBarRequester: FocusRequester,
    isLoading: Boolean,
    isPaginating: Boolean
) {
    val initialIndex = remember(filterState) { maxOf(0, viewModel.getFocusIndex(filterState.genre ?: "Wszystkie") - 2) }
    val listState = rememberLazyGridState(initialFirstVisibleItemIndex = initialIndex)
    val maintenanceRequester = remember { FocusRequester() }

    // Stan opóźnionego wyświetlania komunikatu o braku danych (Debounce)
    var showEmptyState by remember { mutableStateOf(false) }

    // Śledzenie ostatniego fokusu
    var lastFocusedIndex by remember(filterState) {
        mutableIntStateOf(viewModel.getFocusIndex(filterState.genre ?: "Wszystkie"))
    }

    val years = remember { listOf("Wszystkie lata") + (2026 downTo 1990).map { it.toString() } }
    var isSidebarExpanded by remember { mutableStateOf(false) }

    // Efekt uboczny zarządzający stanem pustym
    LaunchedEffect(movies.isEmpty(), isLoading) {
        if (movies.isEmpty() && !isLoading) {
            delay(250)
            showEmptyState = true
        } else {
            showEmptyState = false
        }
    }

    // Obsługa zmiany filtrów
    LaunchedEffect(filterState) {
        lastFocusedIndex = 0
        listState.scrollToItem(0)
    }

    // Reaktwyny mechanizm ładowania (Paginacja) z użyciem snapshotFlow
    LaunchedEffect(listState, movies.size, isLoading, isPaginating, isSidebarExpanded) {
        androidx.compose.runtime.snapshotFlow { listState.layoutInfo.visibleItemsInfo.lastOrNull()?.index }
            .collect { lastIndex ->
                if (lastIndex != null && lastIndex >= movies.size - 12) {
                    if (!isLoading && !isPaginating && !isSidebarExpanded) {
                        onLoadMore()
                    }
                }
            }
    }

    Box(modifier = Modifier.fillMaxSize()) {
        // Główna logika UI zastępująca Crossfade (stabilizacja fokusu)
        if (isLoading && movies.isEmpty()) {
            LazyVerticalGrid(
                columns = GridCells.Adaptive(150.dp),
                contentPadding = PaddingValues(start = 100.dp, end = 50.dp, top = 80.dp, bottom = 16.dp),
                horizontalArrangement = Arrangement.spacedBy(16.dp),
                verticalArrangement = Arrangement.spacedBy(16.dp),
                modifier = Modifier
                    .fillMaxSize()
                    .focusRequester(contentFocusRequester)
                    .focusProperties {
                        up = topBarRequester
                        exit = { direction -> if (direction == androidx.compose.ui.focus.FocusDirection.Exit) androidx.compose.ui.focus.FocusRequester.Cancel else androidx.compose.ui.focus.FocusRequester.Default }
                    },
                userScrollEnabled = false
            ) {
                items(20) { SkeletonMovieCard() }
            }
        } else if (showEmptyState) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .focusRequester(contentFocusRequester)
                    .focusProperties { up = topBarRequester },
                contentAlignment = Alignment.Center
            ) {
                Text(text = "Brak pozycji spełniających kryteria.", color = Color.Gray)
            }
        } else {
            LazyVerticalGrid(
                state = listState,
                columns = GridCells.Adaptive(150.dp),
                contentPadding = PaddingValues(start = 100.dp, end = 50.dp, top = 80.dp, bottom = 16.dp),
                horizontalArrangement = Arrangement.spacedBy(16.dp),
                verticalArrangement = Arrangement.spacedBy(16.dp),
                modifier = Modifier
                    .fillMaxSize()
                    .focusProperties {
                        exit = { direction -> if (direction == androidx.compose.ui.focus.FocusDirection.Exit) androidx.compose.ui.focus.FocusRequester.Cancel else androidx.compose.ui.focus.FocusRequester.Default }
                    }
            ) {
                itemsIndexed(
                    items = movies,
                    key = { _, movie -> movie.originalUrl },
                    contentType = { _, _ -> "movie_card" }
                ) { index, movie ->
                    Box(
                        modifier = Modifier
                            .then(
                                if (index == lastFocusedIndex) {
                                    Modifier
                                        .focusRequester(contentFocusRequester)
                                        .focusRequester(maintenanceRequester)
                                } else Modifier
                            )
                            .onFocusChanged { focusState ->
                                if (focusState.isFocused) {
                                    lastFocusedIndex = index
                                    viewModel.onMovieFocused(filterState.genre ?: "Wszystkie", index)
                                }
                            }
                    ) {
                        CinematicMovieCard(
                            movie = movie,
                            onClick = { onMovieClicked(movie) },
                            onLongFocus = { },
                            focusRequester = null
                        )
                    }
                }
                item(span = { GridItemSpan(maxLineSpan) }) {
                    Spacer(modifier = Modifier.height(100.dp))
                }
            }
        }

        // Nakładka przyciemniająca dla sidebaru
        if (isSidebarExpanded) {
            Box(modifier = Modifier
                .fillMaxSize()
                .background(Color.Black.copy(alpha = 0.7f))
                .zIndex(9f)
            )
        }

        // Panel boczny filtrów

            topBarRequester = topBarRequester,
            contentFocusRequester = contentFocusRequester,
            modifier = Modifier.zIndex(10f)
        )

            when {
                coordinate.y == 0 -> topBarRequester.safeRequestFocus()
                coordinate.y == 1 -> heroFocusRequester.safeRequestFocus()
                coordinate.y >= 2 -> {
                    // Wyliczamy, czy musimy zmienić "stronę" (kategorię)
                    val targetCatIndex = coordinate.y - 2
                    if (targetCatIndex in categories.indices && targetCatIndex != currentCategoryIndex) {
                        currentCategoryIndex = targetCatIndex
                    }
                    delay(100)
                    listFocusRequester.safeRequestFocus()
                }
            }
        }
    }

    // --- INTEGRACJA PRZYCISKU WSTECZ (BACK ANCHOR) ---
    BackHandler(enabled = isHomeTab && (currentY > 0 || currentCategoryIndex > 0)) {
        if (!viewModel.handleBackPress()) {
            // Jeśli VM nie skonsumował (bo jesteśmy na TopBar), sprawdź kategorię
            if (currentCategoryIndex > 0) {
                currentCategoryIndex = 0
                heroFocusRequester.safeRequestFocus()
            }
        }
    }

    // Auto-Carousel logic
    LaunchedEffect(currentMovies, lastInteractionTime) {
        while (true) {
            delay(30000)
            val currentTime = System.currentTimeMillis()
            if (currentTime - lastInteractionTime >= 30000 && currentMovies.isNotEmpty()) {
                val currentIndex = currentMovies.indexOfFirst { it.originalUrl == focusedMovie?.originalUrl }
                val nextIndex = if (currentIndex == -1) 0 else (currentIndex + 1) % currentMovies.size
                val nextMovie = currentMovies[nextIndex]

                focusedMovie = nextMovie
                onMovieFocusedDetails(nextMovie)
            }
        }
    }

    LaunchedEffect(currentCategoryIndex) {
        focusedMovie = moviesByGenre[categories[currentCategoryIndex]]?.firstOrNull()
    }

    LaunchedEffect(effectiveFocusedMovie) {
        effectiveFocusedMovie?.let { current ->
            if (current.overview == null) {
                delay(800)
                onMovieFocusedDetails(current)
            }
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .onPreviewKeyEvent { event ->
                if (event.type == KeyEventType.KeyDown) {
                    lastInteractionTime = System.currentTimeMillis()
                }
                false
            }
    ) {
        ImmersiveBackdrop(movie = effectiveFocusedMovie)

        Column(modifier = Modifier.fillMaxSize()) {
            // SEKCJA HERO (Y = 1, X = 0)
            Box(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
                    .padding(start = 50.dp, top = 100.dp),
                contentAlignment = Alignment.TopStart
            ) {
                if (effectiveFocusedMovie != null) {
                    HeroMovieInfo(
                        movie = effectiveFocusedMovie!!,
                        viewModel = viewModel,
                        focusRequester = heroFocusRequester,
                        topBarRequester = topBarRequester,
                        listFocusRequester = listFocusRequester,
                        onPlay = { onPlayMovie(effectiveFocusedMovie!!) }
                    )
                }
            }

            // SEKCJA LISTY KATEGORII (Y = 2+)
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .wrapContentHeight()
            ) {
                AnimatedContent(
                    targetState = currentCategoryIndex,
                    transitionSpec = {
                        val direction = if (targetState > initialState) 1 else -1
                        (slideInVertically { height -> direction * height } + fadeIn()) with
                                (fadeOut())
                    },
                    label = "CategoryChangeAnim"
                ) { targetIndex ->
                    val safeIndex = targetIndex.coerceIn(categories.indices)
                    val category = categories[safeIndex]
                    val moviesList = moviesByGenre[category] ?: emptyList()

                    SwappableCategoryList(
                        title = category,
                        movies = moviesList,
                        yIndex = 2 + safeIndex,
                        viewModel = viewModel,
                        onMovieFocused = {
                            focusedMovie = it
                            lastInteractionTime = System.currentTimeMillis()
                        },
                        onMovieClicked = onPlayMovie,
                        listStartFocusRequester = if (safeIndex == currentCategoryIndex) listFocusRequester else FocusRequester(),
                        onNavigateUp = {
                            topBarRequester.safeRequestFocus()
                        },
                        onNavigateDown = {
                            if (currentCategoryIndex < categories.lastIndex) {
                                currentCategoryIndex++
                                true
                            } else false
                        }
                    )
                }
            }
        }
    }
}

@Composable
private fun SwappableCategoryList(
    title: String,
    movies: List<MovieItem>,
    yIndex: Int,
    viewModel: MainViewModel,
    onMovieFocused: (MovieItem) -> Unit,
    onMovieClicked: (MovieItem) -> Unit,
    onNavigateUp: () -> Unit,
    onNavigateDown: () -> Boolean,
    listStartFocusRequester: FocusRequester
) {
    val configuration = LocalConfiguration.current
    val screenWidth = configuration.screenWidthDp.dp
    val isMobile = screenWidth < 600.dp

    val cardWidth = when {
        screenWidth < 600.dp -> 100.dp
        screenWidth < 900.dp -> 130.dp
        else -> 150.dp
    }
    val cardHeight = cardWidth * 1.5f

    // Pamiętamy pozycję wiersza wewnątrz kategorii
    val initialIndex = remember(title) { viewModel.getFocusIndex(title) }
    val listState = rememberLazyListState(initialFirstVisibleItemIndex = initialIndex)
    var lastFocusedUrl by rememberSaveable(title) { mutableStateOf<String?>(null) }

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .wrapContentHeight()
            .onPreviewKeyEvent { event ->
                if (event.type == KeyEventType.KeyDown) {
                    when (event.key) {
                        Key.DirectionDown -> onNavigateDown()
                        Key.DirectionUp -> {
                            onNavigateUp()
                            true
                        }
                        else -> false
                    }
                } else false
            }
    ) {
        Text(
            text = title,
            style = MaterialTheme.typography.headlineSmall,
            fontWeight = FontWeight.Bold,
            color = Color(0xFFE0E0E0),
            modifier = Modifier.padding(start = if (isMobile) 16.dp else 50.dp, bottom = 24.dp)
        )

        LazyRow(
            state = listState,
            contentPadding = PaddingValues(
                start = if (isMobile) 16.dp else 50.dp,
                end = if (isMobile) 16.dp else 50.dp,
                bottom = 50.dp
            ),
            horizontalArrangement = Arrangement.spacedBy(if (isMobile) 12.dp else 20.dp),
            modifier = Modifier
                .fillMaxWidth()
                .focusProperties {
                    exit = { direction ->
                        if (direction == androidx.compose.ui.focus.FocusDirection.Exit) androidx.compose.ui.focus.FocusRequester.Cancel
                        else androidx.compose.ui.focus.FocusRequester.Default
                    }
                }
        ) {
            itemsIndexed(
                items = movies,
                key = { _, movie -> movie.originalUrl }
            ) { index, movie ->
                Box(
                    modifier = Modifier
                        .onFocusChanged {
                            if (it.isFocused) {
                                lastFocusedUrl = movie.originalUrl
                                // Raportujemy fokus (yIndex to 2, 3...)
                                viewModel.onMovieFocused(title, index, movie)
                                onMovieFocused(movie)
                            }
                        }
                ) {
                    val isTarget = if (lastFocusedUrl != null) movie.originalUrl == lastFocusedUrl else index == initialIndex

                    CinematicMovieCard(
                        movie = movie,
                        onClick = { onMovieClicked(movie) },
                        onLongFocus = { onMovieFocused(movie) },
                        focusRequester = if (isTarget) listStartFocusRequester else null,
                        cardWidth = cardWidth,
                        cardHeight = cardHeight
                    )
                }
            }
        }
    }
}

@Composable
private fun ImmersiveBackdrop(movie: MovieItem?) {
    Crossfade(
        targetState = movie,
        animationSpec = tween(durationMillis = 800),
        label = "BackdropAnim"
    ) { currentMovie ->
        Box(modifier = Modifier.fillMaxSize()) {
            if (currentMovie?.backdropUrl != null || currentMovie?.thumbnail != null) {
                AsyncImage(
                    model = ImageRequest.Builder(LocalContext.current)
                        .data(currentMovie.backdropUrl ?: currentMovie.thumbnail)
                        .size(1920, 1080)
                        .crossfade(true)
                        .build(),
                    contentDescription = null,
                    contentScale = ContentScale.Crop,
                    modifier = Modifier.fillMaxSize().graphicsLayer { alpha = 0.5f }
                )
            } else {
                Box(modifier = Modifier.fillMaxSize().background(BackgroundDark))
            }

            Box(
                modifier = Modifier.fillMaxSize().background(
                    Brush.verticalGradient(
                        colors = listOf(Color.Transparent, BackgroundDark),
                        startY = 0f, endY = Float.POSITIVE_INFINITY
                    )
                )
            )
            Box(
                modifier = Modifier.fillMaxSize().background(
                    Brush.horizontalGradient(
                        colors = listOf(BackgroundDark.copy(alpha = 0.9f), Color.Transparent),
                        startX = 0f, endX = 1400f
                    )
                )
            )
        }
    }
}

@SuppressLint("SetJavaScriptEnabled")
@Composable
fun LoginScreen(
    onLoginSuccess: (String) -> Unit,
    onBack: () -> Unit
) {
    val loginUrl = "https://filman.cc/logowanie"
    val successUrl = "https://filman.cc/"
    val userAgent = "Mozilla/5.0 (Linux; Android 10; K) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Mobile Safari/537.36"

    var isLoading by remember { mutableStateOf(true) }
    var webViewInstance by remember { mutableStateOf<WebView?>(null) }

    // --- STAN WIRTUALNEGO KURSORA ---
    var cursorX by remember { mutableFloatStateOf(500f) }
    var cursorY by remember { mutableFloatStateOf(500f) }
    var webViewWidth by remember { mutableIntStateOf(1000) }
    var webViewHeight by remember { mutableIntStateOf(1000) }
    val cursorSpeed = 35f // Prędkość poruszania się kursora (px na kliknięcie)
    val scrollThreshold = 100f // Margines od krawędzi wyzwalający scroll
    val scrollAmount = 250 // O ile pikseli scrollować stronę

    // NOWE: Mechanizm wymuszania fokusu dla kontenera przechwytującego klawisze
    val focusRequester = remember { FocusRequester() }
    LaunchedEffect(Unit) {
        try {
            focusRequester.requestFocus()
        } catch (e: Exception) {
            Log.e("LoginScreen", "Focus request failed", e)
        }
    }

    BackHandler {
        if (webViewInstance?.canGoBack() == true) {
            webViewInstance?.goBack()
        } else {
            onBack()
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
            // NOWE: Dodanie fokusu, by Box mógł przechwytywać zdarzenia z klawiatury/pilota
            .focusRequester(focusRequester)
            .focusable()
            // Pobranie wymiarów ekranu, by zablokować wyjście kursora poza ekran
            .onGloballyPositioned { coordinates ->
                webViewWidth = coordinates.size.width
                webViewHeight = coordinates.size.height
            }
            // PRZECHWYTYWANIE KLAWISZY PILOTA
            .onPreviewKeyEvent { event ->
                if (event.type == KeyEventType.KeyDown) {
                    when (event.key) {
                        Key.DirectionUp -> {
                            cursorY -= cursorSpeed
                            if (cursorY < scrollThreshold) {
                                cursorY = scrollThreshold
                                webViewInstance?.scrollBy(0, -scrollAmount)
                            }
                            true
                        }
                        Key.DirectionDown -> {
                            cursorY += cursorSpeed
                            if (cursorY > webViewHeight - scrollThreshold) {
                                cursorY = webViewHeight - scrollThreshold
                                webViewInstance?.scrollBy(0, scrollAmount)
                            }
                            true
                        }
                        Key.DirectionLeft -> {
                            cursorX = (cursorX - cursorSpeed).coerceAtLeast(0f)
                            true
                        }
                        Key.DirectionRight -> {
                            cursorX = (cursorX + cursorSpeed).coerceAtMost(webViewWidth.toFloat())
                            true
                        }
                        Key.DirectionCenter, Key.Enter, Key.NumPadEnter -> {
                            // SYMULACJA KLIKNIĘCIA W MIEJSCU KURSORA
                            webViewInstance?.let { wv ->
                                wv.requestFocus() // <--- DODANE: Wymuszenie fokusu na WebView, by system wiedział, gdzie wpisywać tekst
                                val time = SystemClock.uptimeMillis()
                                val downEvent = MotionEvent.obtain(time, time, MotionEvent.ACTION_DOWN, cursorX, cursorY, 0)
                                val upEvent = MotionEvent.obtain(time, time + 50, MotionEvent.ACTION_UP, cursorX, cursorY, 0)
                                wv.dispatchTouchEvent(downEvent)
                                wv.dispatchTouchEvent(upEvent)
                                downEvent.recycle()
                                upEvent.recycle()
                            }
                            true
                        }
                        else -> false
                    }
                } else {
                    // Konsumujemy KeyUp dla użytych klawiszy, by nie wywoływać podwójnych zdarzeń
                    event.key in listOf(Key.DirectionUp, Key.DirectionDown, Key.DirectionLeft, Key.DirectionRight, Key.DirectionCenter, Key.Enter, Key.NumPadEnter)
                }
            }
    ) {
        AndroidView(
            modifier = Modifier.fillMaxSize(),
            factory = { context ->
                WebView(context).apply {
                    webViewInstance = this

                    setBackgroundColor(android.graphics.Color.BLACK)
                    layoutParams = ViewGroup.LayoutParams(MATCH_PARENT, MATCH_PARENT)

                    // Przywracamy fokus, by Android pozwalał na wysunięcie klawiatury (nasz parent Box i tak przechwytuje strzałki)
                    isFocusable = true
                    isFocusableInTouchMode = true

                    settings.apply {
                        javaScriptEnabled = true
                        domStorageEnabled = true
                        userAgentString = userAgent
                        loadWithOverviewMode = true
                        useWideViewPort = true
                        setSupportZoom(false)
                        builtInZoomControls = false
                    }

                    webViewClient = object : WebViewClient() {
                        override fun onPageStarted(view: WebView?, url: String?, favicon: Bitmap?) {
                            super.onPageStarted(view, url, favicon)
                            isLoading = true
                            // Ustaw kursor na środku po wczytaniu nowej strony
                            cursorX = webViewWidth / 2f
                            cursorY = webViewHeight / 2f
                        }

                        override fun onPageFinished(view: WebView?, url: String?) {
                            super.onPageFinished(view, url)
                            isLoading = false

                            if (url != null && (url == successUrl || url.endsWith("filman.cc/") || !url.contains("/logowanie"))) {
                                val cookies = CookieManager.getInstance().getCookie(url)
                                if (cookies != null && cookies.contains("PHPSESSID")) {
                                    onLoginSuccess(cookies)
                                }
                            }
                        }

                        override fun onReceivedError(view: WebView?, request: WebResourceRequest?, error: WebResourceError?) {
                            super.onReceivedError(view, request, error)
                            if (request?.isForMainFrame == true) {
                                Log.e("LoginScreen", "onReceivedError: ${error?.description} (${error?.errorCode})")
                            }
                        }

                        override fun onReceivedSslError(view: WebView?, handler: SslErrorHandler?, error: SslError?) {
                            handler?.proceed()
                        }

                        override fun shouldOverrideUrlLoading(view: WebView?, url: String?): Boolean {
                            return false
                        }
                    }

                    loadUrl(loginUrl)
                }
            }
        )

        // --- RYSOWANIE WIRTUALNEGO KURSORA ---
        if (!isLoading) {
            Box(
                modifier = Modifier
                    .offset { IntOffset(cursorX.roundToInt(), cursorY.roundToInt()) }
                    // Przesunięcie o połowę rozmiaru, by wskaźnik był wycentrowany na współrzędnych
                    .offset(x = (-12).dp, y = (-12).dp)
                    .size(24.dp)
                    .shadow(8.dp, CircleShape)
                    .background(AccentMint.copy(alpha = 0.8f), CircleShape)
                    .border(2.dp, Color.White, CircleShape)
            ) {
                // Mały punkt precyzyjny w samym środku
                Box(
                    modifier = Modifier
                        .size(4.dp)
                        .background(Color.Black, CircleShape)
                        .align(Alignment.Center)
                )
            }
        }

        // Overlay ładowania
        if (isLoading) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(Color.Black.copy(alpha = 0.5f)),
                contentAlignment = Alignment.Center
            ) {
                CircularProgressIndicator(
                    color = AccentMint,
                    modifier = Modifier.size(48.dp)
                )
            }
        }
    }
}

@OptIn(UnstableApi::class)
@Composable
fun PlayerScreen(
    streamUrl: String,
    originalUrl: String,
    isWebSource: Boolean = false,
    currentMovie: MovieItem? = null,
    nextEpisode: MovieItem? = null,
    isFavorite: Boolean = false,
    onToggleFavorite: () -> Unit = {},
    onPreloadNext: () -> Unit = {},
    onPlayNext: () -> Unit,
    onSaveProgress: (position: Long, duration: Long) -> Unit,
    onBackPressed: () -> Unit
) {
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    val activity = context as? Activity

    // --- STATES ---
    var isUiVisible by remember { mutableStateOf(false) }
    var isTopBarFocused by remember { mutableStateOf(false) }
    var isPlaying by remember { mutableStateOf(true) }
    var currentTime by remember { mutableLongStateOf(0L) }
    var totalDuration by remember { mutableLongStateOf(0L) }
    var centerIcon by remember { mutableStateOf<ImageVector?>(null) }

    var showResumeDialog by remember(streamUrl) { mutableStateOf(false) }
    var savedPosition by remember { mutableLongStateOf(0L) }
    var showNextEpisodeOverlay by remember(streamUrl) { mutableStateOf(false) }
    var playerError by remember(streamUrl) { mutableStateOf<String?>(null) }
    var lastKeyEventTime by remember { mutableLongStateOf(0L) }

    // --- DEBOUNCED SEEKING STATES ---
    var targetSeekPosition by remember { mutableLongStateOf(0L) }
    var isDebounceSeeking by remember { mutableStateOf(false) }
    var seekJob by remember { mutableStateOf<Job?>(null) }

    // --- PROGRESSIVE SEEKING STATES ---
    var consecutiveSeeks by remember { mutableIntStateOf(0) }
    var lastSeekClickTime by remember { mutableLongStateOf(0L) }

    var isDpadSeeking by remember { mutableStateOf(false) }
    val sliderScale by animateFloatAsState(
        targetValue = if (isDebounceSeeking || isDpadSeeking) 1.2f else 1.0f,
        label = "SliderSeekingScale"
    )

    // --- REQUESTERS ---
    val rootFocusRequester = remember { FocusRequester() }
    val backButtonFocusRequester = remember { FocusRequester() }
    val nextEpisodeFocusRequester = remember { FocusRequester() }
    val favoriteFocusRequester = remember { FocusRequester() }
    val resumeBtnFocus = remember { FocusRequester() }
    val nextBtnFocus = remember { FocusRequester() }
    var uiHideJob by remember { mutableStateOf<Job?>(null) }

    val hasValidNextEpisode = nextEpisode != null && nextEpisode.originalUrl != "playback_end"

    // --- EXOPLAYER SETUP ---
    val exoPlayer = remember {
        val audioAttributes = AudioAttributes.Builder()
            .setUsage(C.USAGE_MEDIA)
            .setContentType(C.AUDIO_CONTENT_TYPE_MOVIE)
            .build()

        val renderersFactory = DefaultRenderersFactory(context)
            .setEnableDecoderFallback(true)
            .setExtensionRendererMode(DefaultRenderersFactory.EXTENSION_RENDERER_MODE_ON)

        val trackSelector = DefaultTrackSelector(context).apply {
            parameters = buildUponParameters()
                .setTunnelingEnabled(true)
                .build()
        }

        val loadControl = DefaultLoadControl.Builder()
            .setBufferDurationsMs(15000, 30000, 1500, 2000)
            .build()

        ExoPlayer.Builder(context)
            .setRenderersFactory(renderersFactory)
            .setTrackSelector(trackSelector)
            .setLoadControl(loadControl)
            .setMediaSourceFactory(DefaultMediaSourceFactory(OkHttpDataSource.Factory(NetworkModule.getHttpClient(context))))
            .setAudioAttributes(audioAttributes, true)
            .setHandleAudioBecomingNoisy(true)
            .setSeekBackIncrementMs(10000)
            .setSeekForwardIncrementMs(10000)
            .setVideoChangeFrameRateStrategy(C.VIDEO_CHANGE_FRAME_RATE_STRATEGY_ONLY_IF_SEAMLESS)
            .build()
            .apply {
                playWhenReady = true
                videoScalingMode = C.VIDEO_SCALING_MODE_SCALE_TO_FIT
                setSeekParameters(SeekParameters.EXACT)
            }
    }

    val saveAndExit = {
        if (!isWebSource) {
            onSaveProgress(exoPlayer.currentPosition, exoPlayer.duration)
        }
        onBackPressed()
    }

    fun showUi(autoHide: Boolean = true, keepRootFocus: Boolean = true) {
        if (isWebSource) return
        
        val wasVisible = isUiVisible
        isUiVisible = true
        
        // Zabezpieczenie przed kradzieżą fokusu przez nowo pojawiające się elementy TopBar (focusable(isUiVisible))
        if (!wasVisible && keepRootFocus) {
            scope.launch {
                delay(50)
                rootFocusRequester.safeRequestFocus()
            }
        }
        
        uiHideJob?.cancel()
        if (autoHide && exoPlayer.isPlaying) {
            uiHideJob = scope.launch {
                delay(5000)
                isUiVisible = false
                centerIcon = null
            }
        }
    }

    fun triggerActionIcon(icon: ImageVector) {
        centerIcon = icon
        showUi()
        scope.launch {
            delay(600)
            if (centerIcon == icon) centerIcon = null
        }
    }

    LaunchedEffect(isUiVisible) {
        if (!isUiVisible) rootFocusRequester.safeRequestFocus()
    }

    DisposableEffect(streamUrl, isWebSource) {
        val window = activity?.window
        if (window != null) {
            val insetsController = WindowCompat.getInsetsController(window, window.decorView)
            insetsController.systemBarsBehavior = WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
            insetsController.hide(WindowInsetsCompat.Type.systemBars())
        }

        val playbackManager = PlaybackManager(context)
        savedPosition = playbackManager.getProgress(originalUrl)
        if (savedPosition > 10000 && !isWebSource) {
            showResumeDialog = true
            exoPlayer.playWhenReady = false
        }

        if (!isWebSource) {
            exoPlayer.stop()
            exoPlayer.clearMediaItems()
            val mediaItem = androidx.media3.common.MediaItem.fromUri(streamUrl)
            exoPlayer.setMediaItem(mediaItem)
            exoPlayer.prepare()
            exoPlayer.playWhenReady = !showResumeDialog
        }

        onDispose {
            activity?.window?.let { WindowCompat.getInsetsController(it, it.decorView).show(WindowInsetsCompat.Type.systemBars()) }
        }
    }

    DisposableEffect(exoPlayer, isWebSource, streamUrl) {
        var listener: Player.Listener? = null
        var timerJob: Job? = null

        if (!isWebSource) {
            listener = object : Player.Listener {
                override fun onPlaybackStateChanged(playbackState: Int) {
                    if (playbackState == Player.STATE_ENDED) {
                        showNextEpisodeOverlay = true
                    }
                    isPlaying = exoPlayer.isPlaying
                }
                override fun onIsPlayingChanged(isPlayingChanged: Boolean) {
                    isPlaying = isPlayingChanged
                    if (isPlayingChanged) showUi()
                }
                override fun onPlayerError(error: PlaybackException) {
                    playerError = "Błąd odtwarzania: ${error.errorCodeName} - ${error.message}"
                }
            }
            listener?.let { exoPlayer.addListener(it) }

            timerJob = scope.launch {
                var hasTriggeredPreload = false
                while (true) {
                    if (!isDebounceSeeking) {
                        currentTime = exoPlayer.currentPosition.coerceAtLeast(0L)
                    }
                    val newDuration = exoPlayer.duration

                    if (newDuration != C.TIME_UNSET && newDuration > 0) {
                        totalDuration = newDuration.coerceAtLeast(0L)
                    }

                    if (totalDuration > 0) {
                        if (currentTime >= 5000 && !hasTriggeredPreload) {
                            hasTriggeredPreload = true
                            onPreloadNext()
                        }

                        if (totalDuration - currentTime <= 5000 && nextEpisode != null) {
                            showNextEpisodeOverlay = true
                        }
                    }
                    delay(500)
                }
            }
        }

        onDispose {
            listener?.let { exoPlayer.removeListener(it) }
            timerJob?.cancel()
            exoPlayer.stop()
        }
    }

    BackHandler {
        when {
            showNextEpisodeOverlay -> showNextEpisodeOverlay = false
            isUiVisible && !showResumeDialog && !isWebSource -> {
                isUiVisible = false
            }
            else -> saveAndExit()
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
            .clickable(
                interactionSource = remember { MutableInteractionSource() },
                indication = null
            ) {
                if (!isWebSource && !showResumeDialog && playerError == null) {
                    if (isUiVisible) {
                        isUiVisible = false
                    } else {
                        showUi()
                    }
                }
            }
            .onKeyEvent { event ->
                if (isWebSource || showResumeDialog || playerError != null) return@onKeyEvent false

                if (event.type == KeyEventType.KeyUp) {
                    when (event.key) {
                        Key.DirectionLeft, Key.DirectionRight, Key.MediaRewind, Key.MediaFastForward -> {
                            consecutiveSeeks = 0
                            lastSeekClickTime = 0L
                            exoPlayer.setPlaybackSpeed(1f)
                            return@onKeyEvent true
                        }
                        Key.DirectionCenter, Key.Enter, Key.MediaPlayPause, Key.DirectionUp, Key.DirectionDown -> {
                            exoPlayer.setPlaybackSpeed(1f)
                            return@onKeyEvent true
                        }
                    }
                    return@onKeyEvent false
                }

                if (event.type != KeyEventType.KeyDown) return@onKeyEvent false

                if (isTopBarFocused) {
                    if (event.key == Key.DirectionDown) {
                        rootFocusRequester.safeRequestFocus()
                        return@onKeyEvent true
                    }
                    return@onKeyEvent false
                }

                val currentTimeMillis = System.currentTimeMillis()

                when (event.key) {
                    Key.DirectionLeft, Key.MediaRewind, Key.DirectionRight, Key.MediaFastForward -> {
                        // Rate limiting to handle Android TV auto-repeat
                        if (currentTimeMillis - lastSeekClickTime < 150 && lastSeekClickTime != 0L) return@onKeyEvent true

                        isDebounceSeeking = true

                        // Reset acceleration if more than 800ms passed since last seek
                        if (currentTimeMillis - lastSeekClickTime > 800) {
                            consecutiveSeeks = 0
                        }
                        consecutiveSeeks++
                        lastSeekClickTime = currentTimeMillis

                        // Progressive seeking step logic
                        val seekStep = when {
                            consecutiveSeeks <= 3 -> 10_000L
                            consecutiveSeeks <= 8 -> 30_000L
                            else -> 60_000L
                        }

                        if (seekJob?.isActive != true) {
                            targetSeekPosition = exoPlayer.currentPosition
                        }

                        if (event.key == Key.DirectionLeft || event.key == Key.MediaRewind) {
                            targetSeekPosition = (targetSeekPosition - seekStep).coerceAtLeast(0L)
                            triggerActionIcon(if (seekStep >= 60000L) Icons.Rounded.FastRewind else Icons.Rounded.Replay10)
                        } else {
                            targetSeekPosition = (targetSeekPosition + seekStep).coerceAtMost(totalDuration)
                            triggerActionIcon(if (seekStep >= 60000L) Icons.Rounded.FastForward else Icons.Rounded.Forward10)
                        }

                        currentTime = targetSeekPosition
                        seekJob?.cancel()
                        seekJob = scope.launch {
                            // 300ms delay to allow UI to move smoothly before actual seek
                            delay(300)
                            exoPlayer.seekTo(targetSeekPosition)
                            isDebounceSeeking = false
                        }
                        true
                    }
                    Key.DirectionCenter, Key.Enter, Key.NumPadEnter, Key.Spacebar, Key.MediaPlayPause -> {
                        if (exoPlayer.isPlaying) exoPlayer.pause() else exoPlayer.play()
                        triggerActionIcon(if (exoPlayer.isPlaying) Icons.Rounded.Pause else Icons.Rounded.PlayArrow)
                        true
                    }
                    Key.DirectionUp -> {
                        showUi(autoHide = true, keepRootFocus = false)
                        backButtonFocusRequester.safeRequestFocus()
                        true
                    }
                    Key.DirectionDown -> {
                        if (!isUiVisible) {
                            showUi(true)
                            true
                        } else {
                            isUiVisible = false
                            true
                        }
                    }
                    else -> {
                        if (currentTimeMillis - lastKeyEventTime < 150) return@onKeyEvent true
                        lastKeyEventTime = currentTimeMillis
                        false
                    }
                }
            }
            .focusRequester(rootFocusRequester)
            .focusable()
    ) {
        if (isWebSource) {
            AndroidView(
                factory = { ctx ->
                    WebView(ctx).apply {
                        setBackgroundColor(AndroidColor.BLACK)
                        settings.javaScriptEnabled = true
                        settings.domStorageEnabled = true
                        settings.mixedContentMode = WebSettings.MIXED_CONTENT_ALWAYS_ALLOW
                        webViewClient = WebViewClient()
                        loadUrl(streamUrl)
                    }
                },
                modifier = Modifier.fillMaxSize().zIndex(0f)
            )
        } else {
            AndroidView(
                modifier = Modifier.fillMaxSize().zIndex(0f),
                factory = { ctx ->
                    PlayerView(ctx).apply {
                        player = exoPlayer
                        useController = false
                        resizeMode = AspectRatioFrameLayout.RESIZE_MODE_FIT
                        keepScreenOn = true
                        isFocusable = false
                        isFocusableInTouchMode = false
                        findViewById<View>(androidx.media3.ui.R.id.exo_shutter)?.setBackgroundColor(AndroidColor.TRANSPARENT)
                    }
                }
            )

            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .zIndex(1f)
                    .alpha(if (isUiVisible) 1f else 0f)
            ) {
                // Top Navigation Bar
                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(24.dp)
                        .align(Alignment.TopStart)
                        .onFocusChanged { isTopBarFocused = it.hasFocus },
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    val backInteractionSource = remember { MutableInteractionSource() }
                    val isBackFocused by backInteractionSource.collectIsFocusedAsState()

                    Box(
                        modifier = Modifier
                            .size(48.dp)
                            .background(if (isBackFocused) Color.White else Color.Black.copy(alpha = 0.5f), CircleShape)
                            .focusRequester(backButtonFocusRequester)
                            .focusProperties {
                                up = FocusRequester.Cancel
                                left = FocusRequester.Cancel
                                right = if (hasValidNextEpisode) nextEpisodeFocusRequester else favoriteFocusRequester
                                down = rootFocusRequester
                            }
                            .clickable(interactionSource = backInteractionSource, indication = null, enabled = isUiVisible) {
                                saveAndExit()
                            }
                            .focusable(isUiVisible),
                        contentAlignment = Alignment.Center
                    ) {
                        Icon(
                            imageVector = Icons.Rounded.ArrowBack,
                            contentDescription = "Wróć",
                            tint = if (isBackFocused) Color.Black else Color.White
                        )
                    }

                    if (currentMovie != null) {
                        Text(
                            text = currentMovie.title,
                            color = Color.White,
                            style = MaterialTheme.typography.titleLarge,
                            fontWeight = FontWeight.Bold,
                            maxLines = 1,
                            overflow = TextOverflow.Ellipsis,
                            modifier = Modifier.padding(start = 24.dp)
                        )

                        Spacer(modifier = Modifier.weight(1f))

                        if (hasValidNextEpisode) {
                            val nextInteractionSource = remember { MutableInteractionSource() }
                            val isNextFocused by nextInteractionSource.collectIsFocusedAsState()

                            Box(
                                modifier = Modifier
                                    .size(48.dp)
                                    .background(if (isNextFocused) Color.White else Color.Black.copy(alpha = 0.5f), CircleShape)
                                    .focusRequester(nextEpisodeFocusRequester)
                                    .focusProperties {
                                        up = FocusRequester.Cancel
                                        left = backButtonFocusRequester
                                        right = favoriteFocusRequester
                                        down = rootFocusRequester
                                    }
                                    .clickable(interactionSource = nextInteractionSource, indication = null, enabled = isUiVisible) {
                                        onPlayNext()
                                    }
                                    .focusable(isUiVisible),
                                contentAlignment = Alignment.Center
                            ) {
                                Icon(
                                    imageVector = Icons.Rounded.SkipNext,
                                    contentDescription = "Następny odcinek",
                                    tint = if (isNextFocused) Color.Black else Color.White
                                )
                            }

                            Spacer(modifier = Modifier.width(16.dp))
                        }

                        val favInteractionSource = remember { MutableInteractionSource() }
                        val isFavFocused by favInteractionSource.collectIsFocusedAsState()

                        Box(
                            modifier = Modifier
                                .size(48.dp)
                                .background(if (isFavFocused) Color.White else Color.Black.copy(alpha = 0.5f), CircleShape)
                                .focusRequester(favoriteFocusRequester)
                                .focusProperties {
                                    up = FocusRequester.Cancel
                                    right = FocusRequester.Cancel
                                    left = if (hasValidNextEpisode) nextEpisodeFocusRequester else backButtonFocusRequester
                                    down = rootFocusRequester
                                }
                                .clickable(interactionSource = favInteractionSource, indication = null, enabled = isUiVisible) {
                                    onToggleFavorite()
                                }
                                .focusable(isUiVisible),
                            contentAlignment = Alignment.Center
                        ) {
                            Icon(
                                imageVector = if (isFavorite) Icons.Rounded.Favorite else Icons.Rounded.FavoriteBorder,
                                contentDescription = "Ulubione",
                                tint = if (isFavorite) AccentMint else (if (isFavFocused) Color.Black else Color.White),
                                modifier = Modifier.size(24.dp)
                            )
                        }
                    }
                }

                // Center Action Indicator & Touch Controls
                Box(modifier = Modifier.align(Alignment.Center)) {
                    if (centerIcon != null) {
                        Box(
                            modifier = Modifier.size(84.dp).background(Color.Black.copy(alpha = 0.7f), CircleShape)
                                .border(1.dp, Color.White.copy(alpha = 0.1f), CircleShape),
                            contentAlignment = Alignment.Center
                        ) {
                            Icon(imageVector = centerIcon!!, contentDescription = null, tint = Color.White, modifier = Modifier.size(48.dp))
                        }
                    } else {
                        PlayerCenterControls(
                            isPlaying = isPlaying,
                            onPlayPauseClick = {
                                if (exoPlayer.isPlaying) exoPlayer.pause() else exoPlayer.play()
                                triggerActionIcon(if (exoPlayer.isPlaying) Icons.Rounded.Pause else Icons.Rounded.PlayArrow)
                            },
                            onReplayClick = {
                                val newPosition = (exoPlayer.currentPosition - 10000).coerceAtLeast(0L)
                                exoPlayer.seekTo(newPosition)
                                triggerActionIcon(Icons.Rounded.Replay10)
                            },
                            onForwardClick = {
                                val newPosition = (exoPlayer.currentPosition + 10000).coerceAtMost(totalDuration)
                                exoPlayer.seekTo(newPosition)
                                triggerActionIcon(Icons.Rounded.Forward10)
                            }
                        )
                    }
                }

                PlayerBottomControls(
                    currentTime = currentTime,
                    totalDuration = totalDuration,
                    isUiVisible = isUiVisible,
                    sliderScale = sliderScale,
                    onSeek = { position ->
                        currentTime = position
                        exoPlayer.seekTo(position)
                    }
                )
            }

            if (showResumeDialog) {
                ResumeDialog(
                    savedPosition = savedPosition,
                    onResume = {
                        showResumeDialog = false
                        exoPlayer.seekTo(savedPosition)
                        exoPlayer.play()
                    },
                    onRestart = {
                        showResumeDialog = false
                        exoPlayer.seekTo(0)
                        exoPlayer.play()
                    },
                    focusRequester = resumeBtnFocus
                )
            }

            if (showNextEpisodeOverlay && nextEpisode != null) {
                NextEpisodeOverlay(
                    nextEpisode = nextEpisode,
                    onPlayNext = onPlayNext,
                    onCancel = { showNextEpisodeOverlay = false },
                    onBackPressed = saveAndExit,
                    focusRequester = nextBtnFocus
                )
            }

            if (playerError != null) {
                AlertDialog(
                    onDismissRequest = onBackPressed,
                    title = { Text("Błąd") },
                    text = { Text(playerError!!) },
                    confirmButton = { Button(onClick = onBackPressed) { Text("Wróć") } }
                )
            }
        }
    }

    LaunchedEffect(Unit) {
        rootFocusRequester.requestFocus()
    }
}

@Composable
fun NextEpisodeOverlay(
    nextEpisode: MovieItem,
    onPlayNext: () -> Unit,
    onCancel: () -> Unit,
    onBackPressed: () -> Unit,
    focusRequester: FocusRequester
) {
    var countdown by remember { mutableIntStateOf(5) }
    var hasRequestedFocus by remember { mutableStateOf(false) }
    val isPlaybackEnd = nextEpisode.originalUrl == "playback_end"

    LaunchedEffect(Unit) {
        if (!isPlaybackEnd) {
            while (countdown > 0) {
                delay(1000)
                countdown--
            }
            onPlayNext()
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .zIndex(10f)
            .background(Color.Black.copy(alpha = 0.6f)),
        contentAlignment = Alignment.Center
    ) {
        Card(
            modifier = Modifier
                .width(550.dp)
                .shadow(12.dp, RoundedCornerShape(16.dp)),
            shape = RoundedCornerShape(16.dp),
            colors = CardDefaults.cardColors(containerColor = Color(0xFF0A0A0A))
        ) {
            Column(
                modifier = Modifier.padding(40.dp),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                Text(
                    text = "NASTĘPNY ODCINEK",
                    color = Color.Gray,
                    style = MaterialTheme.typography.labelSmall,
                    letterSpacing = 4.sp,
                    textAlign = TextAlign.Center
                )

                Spacer(modifier = Modifier.height(16.dp))

                Text(
                    text = nextEpisode.title,
                    color = Color.White,
                    style = MaterialTheme.typography.headlineMedium,
                    fontWeight = FontWeight.ExtraBold,
                    textAlign = TextAlign.Center,
                    maxLines = 2,
                    overflow = TextOverflow.Ellipsis
                )

                Spacer(modifier = Modifier.height(24.dp))

                Text(
                    text = if (isPlaybackEnd) "To już wszystko w tej sekcji." else "Automatyczne odtwarzanie za ${countdown}s...",
                    color = AccentMint,
                    style = MaterialTheme.typography.bodyMedium,
                    fontWeight = FontWeight.Bold,
                    textAlign = TextAlign.Center
                )

                Spacer(modifier = Modifier.height(32.dp))

                Row(
                    horizontalArrangement = Arrangement.Center,
                    verticalAlignment = Alignment.CenterVertically,
                    modifier = Modifier.fillMaxWidth()
                ) {
                    if (!isPlaybackEnd) {
                        val playInteraction = remember { MutableInteractionSource() }
                        val isPlayFocused by playInteraction.collectIsFocusedAsState()

                        Surface(
                            modifier = Modifier
                                .focusRequester(focusRequester)
                                .onGloballyPositioned {
                                    if (!hasRequestedFocus) {
                                        focusRequester.safeRequestFocus()
                                        hasRequestedFocus = true
                                    }
                                }
                                .clickable(interactionSource = playInteraction, indication = null) { onPlayNext() }
                                .focusable(interactionSource = playInteraction),
                            shape = RoundedCornerShape(8.dp),
                            color = if (isPlayFocused) Color.White else AccentMint
                        ) {
                            Text(
                                text = "Odtwórz teraz ($countdown)",
                                color = Color.Black,
                                modifier = Modifier.padding(horizontal = 24.dp, vertical = 12.dp),
                                fontWeight = FontWeight.Bold
                            )
                        }

                        Spacer(modifier = Modifier.width(16.dp))
                    }

                    val cancelInteraction = remember { MutableInteractionSource() }
                    val isCancelFocused by cancelInteraction.collectIsFocusedAsState()

                    Surface(
                        modifier = Modifier
                            .then(if (isPlaybackEnd) Modifier.focusRequester(focusRequester) else Modifier)
                            .onGloballyPositioned {
                                if (isPlaybackEnd && !hasRequestedFocus) {
                                    focusRequester.safeRequestFocus()
                                    hasRequestedFocus = true
                                }
                            }
                            .clickable(interactionSource = cancelInteraction, indication = null) {
                                if (isPlaybackEnd) onBackPressed() else onCancel()
                            }
                            .focusable(interactionSource = cancelInteraction)
                            .border(1.dp, Color.DarkGray, RoundedCornerShape(8.dp)),
                        shape = RoundedCornerShape(8.dp),
                        color = if (isCancelFocused) Color.White.copy(alpha = 0.2f) else Color(0xFF1E1E1E)

                            text = if (isPlaybackEnd) "ZAKOŃCZ / WRÓĆ" else "ZOSTAŃ TUTAJ",

                            modifier = Modifier.padding(horizontal = 24.dp, vertical = 12.dp),
                            fontWeight = FontWeight.Bold
                        )
                    }
                }
            }
        }
    }

fun formatTime(millis: Long): String {
    val seconds = (millis / 1000) % 60
    val minutes = (millis / (1000 * 60)) % 60
    val hours = millis / (1000 * 60 * 60)
    return if (hours > 0) "%d:%02d:%02d".format(hours, minutes, seconds) else "%02d:%02d".format(minutes, seconds)
}

@Composable
fun FilmanTVTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    // Wyłączamy dynamiczne kolory domyślnie, aby zachować branding FilmanTV
    dynamicColor: Boolean = false,
    content: @Composable () -> Unit
) {
    val colorScheme = when {
        dynamicColor && Build.VERSION.SDK_INT >= Build.VERSION_CODES.S -> {
            val context = LocalContext.current
            if (darkTheme) dynamicDarkColorScheme(context) else dynamicLightColorScheme(context)
        }

        darkTheme -> DarkColorScheme
        else -> LightColorScheme
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = Typography,
        content = content
    )
}

const fs = require('fs');
const path = require('path');
const chokidar = require('chokidar');

// Domyślna konfiguracja (fallback)
let CONFIG = {
    watchFolder: './',
    outputFile: 'context_bundle.txt',
    allowedExtensions: ['.js', '.jsx', '.ts', '.tsx', '.css', '.html', '.json', '.py', '.md'],
    ignorePatterns: ['node_modules', '.git', 'dist', 'build', 'package-lock.json']
};

try {
    if (fs.existsSync('watcher.config.json')) {
        const userConfig = JSON.parse(fs.readFileSync('watcher.config.json', 'utf8'));
        CONFIG = { ...CONFIG, ...userConfig };
    }
    
    // ZABEZPIECZENIE: Zawsze dodaj plik wyjściowy do ignorowanych
    if (!CONFIG.ignorePatterns.includes(CONFIG.outputFile)) {
        CONFIG.ignorePatterns.push(CONFIG.outputFile);
    }
    
} catch (e) {
    console.warn('⚠️ Nie udało się wczytać configu, używam domyślnego.');
}

let debounceTimer;

const mergeFiles = () => {
    console.log('🔄 Wykryto zmiany. Generowanie kontekstu...');
    let fullContent = `--- PROJECT CONTEXT GENERATED AT ${new Date().toLocaleString()} ---\n\n`;
    
    fullContent += `// --- Configuration used: watcher.config.json ---\n`;
    fullContent += JSON.stringify(CONFIG, null, 2) + '\n\n';

    const walkSync = (dir, filelist = []) => {
        const files = fs.readdirSync(dir);
        files.forEach((file) => {
            const filePath = path.join(dir, file);
            const stat = fs.statSync(filePath);
            
            if (CONFIG.ignorePatterns.some(pattern => filePath.includes(pattern))) return;

            if (stat.isDirectory()) {
                walkSync(filePath, filelist);
            } else {
                if (CONFIG.allowedExtensions.includes(path.extname(file))) {
                    filelist.push(filePath);
                }
            }
        });
        return filelist;
    };

    try {
        const allFiles = walkSync(CONFIG.watchFolder);
        allFiles.forEach(filePath => {
            const content = fs.readFileSync(filePath, 'utf8');
            const relativePath = path.relative('.', filePath).replace(/\\/g, '/');
            fullContent += `\n// ==========================================\n// --- File: ${relativePath} ---\n// ==========================================\n\n${content}\n`;
        });

        fs.writeFileSync(CONFIG.outputFile, fullContent);
        console.log(`✅ Zaktualizowano: ${CONFIG.outputFile} (${allFiles.length} plików)`);
    } catch (err) {
        console.error('❌ Błąd:', err.message);
    }
};

const watcher = chokidar.watch(CONFIG.watchFolder, {
    ignored: (path) => CONFIG.ignorePatterns.some(s => path.includes(s)),
    persistent: true,
    ignoreInitial: true
});

const handleEvent = () => {
    clearTimeout(debounceTimer);
    debounceTimer = setTimeout(() => mergeFiles(), 300);
};

watcher.on('all', handleEvent);

console.log(`👀 Watcher uruchomiony. Obserwuję folder: ${CONFIG.watchFolder}`);
console.log(`📝 Plik wynikowy: ${CONFIG.outputFile}`);
console.log(`💡 Edytuj 'watcher.config.json' aby zmienić ustawienia.`);
mergeFiles();
